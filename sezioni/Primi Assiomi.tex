\section{I primi assiomi}
\subsection{Assiomi dell'insieme vuoto e di estensionalit\`a}
\vskip\baselineskip
\begin{axiom}
[Assioma dell'insieme vuoto]
\label{ax1}
``Esiste un insieme vuoto.''$\drkgrn\;\;\longleftarrow\text{informalmente}$
\[ {\drkgrn\mathllap{\text{questo \`e l'assioma}\longrightarrow\;\;}} \exists x \; \forall y \; y \not\in x \]
\end{axiom}

\begin{note}
Questo assioma non sarebbe strettamente necessario, in quanto potremmo ottenere un insieme vuoto anche come sottoprodotto, per esempio, dell'assioma dell'infinito che vedremo in seguito.
Tuttavia \`e bello poter partire avendo per le mani almeno un insieme.
\end{note}

\begin{axiom}
[Assioma di estensionalit\`a]
\label{ax2}
``Un insieme \`e determinato dalla collezione dei suoi elementi.''\\
\hbox to 2em{\hss}{\drkgrn o anche}\\
``Due insiemi coincidono se e solo se hanno i medesimi elementi.''
\[ \forall a \; \forall b \; a = b \leftrightarrow \forall x (x \in a \leftrightarrow x \in b)
	\]
\end{axiom}

\begin{exercise}
Dimostra che la freccia $a = b \rightarrow \forall x (x \in a
\leftrightarrow x \in b)$, in realt\`a, segue dal fatto che, se $a=b$,
allora $a$ e~$b$ sono inidistinguibili.
\end{exercise}

\textbf{\underline{Convenzione}} Nei contesti meno formali, assumeremo che
le variabili libere, ossia quelle non quantificate, se non specificato
altrimenti, siano quantificate universalmente all'inizio della formula.
Per cui, per esempio, potremmo scrivere
l'assioma di estensionalit\`a semplicemente nella forma
$ a = b \;\leftrightarrow\; \forall x (x \in a \leftrightarrow x \in
b) $, in cui i quantificatori $\forall a$ e~$\forall b$ sono impliciti.

\begin{proposition}
C'\`e un unico insieme vuoto.
\[ \exists ! x \forall y y \not \in x
	\]
\end{proposition}

\begin{proof}
L'esistenza \`e l'\hyperref[ax1]{assioma del vuoto.} Per dimostrare
l'unicit\`a,
consideriamo due insiemi vuoti $x_1$ e $x_2$, ossia supponiamo $\forall y \, y \not\in x_1$, e $\forall y \, y \not \in x_2$. Allora
\[ \forall y (y \in x_1 \leftrightarrow y \in x_2) \]
perch\'e $y \in x_1$ e $y \in x_2$ sono entrambe necessariamente false.
Quindi, per \hyperref[ax2]{estensionalit\`a,} $x_1 = x_2$, che \`e la tesi.
\end{proof}

\begin{proof}[Dimostrazione formale.] Questo livello di pedanteria non \`e
necessario, ma, per una volta, proviamo a dimostrare in ogni dettaglio la
formula $\drkred \exists ! x (\forall y \, y \not \in x)$. Per definizione di
$\oldexists!$, questa formula equivale a
\[\drkred  \exists {x_1} \Bigl((\forall y \, y \not \in x_1) \land
\forall {x_2} \bigl((\forall y \, y \not \in x_2) \rightarrow x_2 =
x_1\bigr)\Bigr)
	\]
Per l'\hyperref[ax1]{assioma del vuoto,} $\exists {x_1} \forall y \, y \not \in x_1$: fissiamo questo $x_1$. Resta da dimostrare che
\[ \drkred (\forall y \, y \not \in x_1) \land \forall {x_2} \bigl((\forall y \, y
\not \in x_2) \rightarrow x_2 = x_1\bigr) \]
Per costruzione, $\drkblu \forall y\,  y \not\in x_1$ \`e vera, quindi resta da
dimostrare
\[ \drkred \forall {x_2} \bigl((\forall y \, y \not \in x_2) \rightarrow x_2 =
x_1\bigr) \]
Ora prendiamo un $x_2$ arbitrario, dobbiamo dimostrare
\[ \drkred (\forall y \, y \not \in x_2) \rightarrow x_2 =  x_1 \]
Assumiamo $\drkblu \forall y\, y \not \in x_2$: dobbiamo ottenere $\drkred x_2 =x_1$.
Ricordiamo che abbiamo assunto anche l'ipotesi $\drkblu \forall y\, y \not \in x_1$. Quindi, preso un
$y$ arbitrario, $y \in x_2$ e $y \in x_1$ sono entrambe false. La tabella
di verit\`a di $\leftrightarrow$ ci dice quindi che vale $\drkblu y \in x_2 \leftrightarrow y \in x_1$, e, per 
l'arbitrariet\`a di $y$
\[ \drkblu \forall y (y \in x_2 \leftrightarrow y \in x_1)
	\]
Dall'\hyperref[ax2]{assioma di estensionalit\`a}
\[ \forall y (y \in x_2 \leftrightarrow y \in x_1) \rightarrow x_2 = x_1
	\]
otteniamo quindi $\drkblu x_2 = x_1.$
\end{proof}

\begin{note}
Questa \`e la seconda e ultima dimostrazione delirante di queste dispense.
Vi sarete convinti che scrivere a un simile livello di dettaglio, in
realt\`a, nasconde il nocciolo dell'argomento, per cui \textbf{non \`e una buona idea.}
\end{note}

\begin{notation}
L'unicit\`a dell'insieme vuoto ci giustifica ad introdurre delle nuove abbreviazioni:
\vskip-.5\baselineskip
\[ {\drkblu x = \emptyset} \;\; \Mydef \;\; \forall y \, y \not\in x \qquad {\drkblu
\emptyset \in x} \;\; \Mydef \;\; \exists z (z = \emptyset \land z \in x)
\qquad {\drkblu x \in \emptyset} \;\; \Mydef \;\; \above{\footnotesize\drkblu=
``falso''\\\vskip-.4\baselineskip$\downarrow$}\bot
	\]
\end{notation}

\begin{note}
Questo procedimento \`e generico: ogniqualvolta dimostreremo che esiste un
unico insieme~$\alpha$ con una certa propriet\`a $\Phi(\alpha)$, potremo associare un
nuovo simbolo~$\Leo$ a questo~$\alpha$. Formalmente, l'operazione si compie
definendo tre abbreviazioni:
\[ {\drkblu x = \Leo} \; \Mydef \; \Phi(x) \quad {\drkblu
\Leo \in x} \; \Mydef \; \exists y (\Phi(y) \land y\in x) \quad
{\drkblu x \in \Leo} \; \Mydef \; \exists y (\Phi(y) \land x\in y) \;\; \]
Stante che il procedimento \`e meccanico, ci limiteremo a enunciare
solo la prima di queste definizioni.
\end{note}

\subsection{Assioma di separazione}
\vskip\baselineskip
\begin{axiom}
[Assioma di separazione]
\label{ax3}
``Se $A$ \`e un insieme, e $\Phi(x)$ una formula insiemistica, allora $\{x \in
A | \Phi (x)\}$ \`e un insieme.''
\[ \forall A \; \exists B \; \forall x \; x \in B \leftrightarrow (x \in A
\land \Phi (x))
	\]
\end{axiom}

\begin{note}
Tecnicamente l'assioma di separazione \`e uno \vocab{schema di assiomi},
ossia una regola che, per ogni possibile formula $\Phi$, ci permette di scrivere un assioma.
\end{note}

\begin{proposition}
Fissati $A$ e $\Phi(x)$, l'insieme $\{x \in A | \Phi(x)\}$ \`e univocamente definito. Ossia:
\[ \forall A \; \exists \textcolor{red}{!} B \; \forall x \; x \in B
\leftrightarrow (x \in A \land \Phi(x))
	\]
\end{proposition}

\begin{proof}
Come per l'unicit\`a dell'insieme vuoto, supponiamo di avere $B_1$ e $B_2$ tali che:
\[ \forall x \, x \in B_1 \leftrightarrow (x \in A \land \Phi(x)) \qquad
\forall x \, x \in B_2 \leftrightarrow (x \in A \land \Phi(x))
	\]
Allora $x \in B_1 \leftrightarrow (x \in A \land \Phi(x))
\leftrightarrow x \in B_2$, quindi, per \hyperref[ax2]{estensionalit\`a}, $B_1 = B_2$.
\end{proof}

\begin{notation}
Vista l'unicit\`a, possiamo introdurre una nuova abbreviazione:
\[ {\drkblu B = \{x \in A | \Phi(x)\}} \;\;\Mydef\;\; \forall x \, x \in B
\leftrightarrow (x \in A \land \Phi(x))
	\]
\end{notation}

Osserviamo che l'assioma di separazione \`e una forma indebolita del
principio di collezione. Riducendo in questo modo il principio di
collezione, si ottiene che il Paradosso di Russell diventa una proposizione.

\begin{proposition}
Non esiste l'insieme di tutti gli insiemi: $\nexists V \; \forall x \;
x \in V$
\end{proposition}

\begin{proof}
Supponiamo, per assurdo, che esista questo $V$. Allora, per
\hyperref[ax3]{separazione} con la formula $\Phi(x) \equiv x \not \in x$, esiste l'insieme
\[ N = \{x \in V | x \not\in x\}
	\]
che, per definizione, ha la propriet\`a
\[ \forall x \, x \in N \leftrightarrow (x \in V \land x \not \in x)
	\]
Per ipotesi assurda, $x \in V$ \`e sempre vera, quindi
\[ \forall x \, x \in N \leftrightarrow x \not\in x
	\]
prendendo, ora, $x = N$, abbiamo $N \in N \leftrightarrow N \not\in N$, assurdo.
\end{proof}

\subsection{Classi e classi proprie}
Sebbene, abbiamo detto, gli unici oggetti della teoria degli insiemi sono gli insiemi, usualmente ci si riferisce alla collezione di tutti gli insiemi 
che soddisfano una certa formula come ad una specie di insieme: una
\vocab{classe}. Pi\`u precisamente, data una formula $\Phi(x)$, se diciamo:
``sia $C$ la classe degli insiemi $x$ tali che $\Phi(x)$''
intendiamo dire che useremo la scrittura $\drkblu x \in C$ come una semplice
abbreviazione per la formula $\Phi(x)$. Si capisce che il glifo~$C$,
qui, non ha un senso proprio: si tratta semplicemente di una parte del
simbolo ``$\in C$'' che \`e un tutt'uno inscindibile.
Non avrebbe senso, per esempio, scrivere $\drkred C \in \text{qualcosa}.$ In altri
termini, se scriviamo $x \in C$ in luogo di $\Phi(x)$, \`e solo come ausilio
all'intuizione -- per comodit\`a insomma -- e sarebbe stato ugualmente
lecito
decidere di scrivere $\mouse x$, o nient'altro
che~$\Phi(x)$. Per caso, il simbolo che abbiamo scelto, $\in C,$
assomiglia graficamente a un simbolo di appartenenza applicato al
glifo~$C$.

\begin{notation}
Detta $C$ la classe degli insiemi~$x$ tali che vale~$\Phi(x)$, definiamo:
\begin{align*}
{\drkblu x \in C} &\;\;\Mydef\;\; \Phi(x) & {\drkblu \forall{x\in
C}\dotsc} &\;\;\Mydef\;\; \forall x \,
\Phi(x)\rightarrow\dotsc\\
&& {\drkblu \exists{x\in C}\dotsc} &\;\;\Mydef\;\; \exists x \,
x\in C \land \dotsc
\end{align*}
\end{notation}

\begin{definition}
La classe $V$ definita dalla formula
\[ {\drkblu x \in V} \;\;\Mydef\;\; x = x \]
si dice \vocab{classe universale,} ed \`e la classe di tutti gli insiemi.
\end{definition}

\begin{note}
Insomma, scrivere $x \in V$ non dice molto: \`e una formula sempre vera.
\end{note}

\begin{notation}[Uguaglianza tra classi]
Date due classi $C$ e $D$, che, ricordiamo, non significa altro che ``date due formule$\ldots$'', definiamo l'abbreviazione:
\[ {\drkblu C = D} \;\;\Mydef\;\; \forall x ((x \in C) \leftrightarrow (x \in D))
\]
\end{notation}

Ora, dato un qualunque insieme $A$, possiamo definire la classe $\hat{A}$
degli $x$ tali che $x \in A$. Se $\hat{A} = \hat{B}$, per definizione
\[ \forall x ((x \in A) \leftrightarrow (x \in B))
	\]
quindi $A = B$ per \hyperref[ax2]{estensionalit\`a}. Ha quindi senso, con un leggero abuso di notazione, omettere il cappelletto $\hat{}$ e identificare la classe $\hat{A}$ semplicemente con $A$. In questo senso,
abbiamo classi che sono insiemi -- formalmente $C$ \`e un insieme se $C =
\hat{A}$ per qualche insieme $A$ -- e classi che non sono insiemi.
Chiamiamo \vocab{classe propria} una classe che non \`e un insieme.

\begin{example}
$V$ \`e una classe propria.
\end{example}

\textbf{\underline{L'intuizione}}, che sar\`a pi\`u chiara via via che procediamo nel corso, \`e che le classi proprie sono troppo grandi per essere insiemi.

\subsection{Assioma del paio e coppia di Kuratowski}
I primi tre assiomi ci dicono, a grandi linee, che, entro i limiti di
quanto si pu\`o fare rinunciando al principio di collezione -- il fatto
che esiste $\{x | \, \text{una qualunque propriet\`a}\}$ -- gli insiemi
sono delle specie di collezioni:
sono determinati dai loro elementi, e li si pu\`o dividere in collezioni pi\`u piccole in maniera arbitraria.

Ci troviamo, adesso, nella necessit\`a di procurarci qualche insieme con cui lavorare. I prossimi assiomi serviranno per giustificare le costruzioni con cui,
usualmente, si definiscono nuovi insiemi. Per esempio, abbiamo bisogno di costruire certi insiemi di base, tipo l'insieme dei numeri interi o insiemi finiti i cui elementi sono elencati esplicitamente, fare prodotti di insiemi esistenti, 
considerare le funzioni fra insiemi esistenti, \etc.

\begin{axiom}
[Assioma del paio]
\label{ax4}
``Dati $a$ e $b$ esiste l'insieme $\{a,b\}$.''
\[ \forall a \; \forall b \; \exists P \; \forall x \; x \in P \leftrightarrow (x = a \lor x = b)
	\]
\end{axiom}

\begin{proposition}
Fissati $a$ e $b$, l'insieme $\{a,b\}$ \`e univocamente determinato.
\[\forall a \; \forall b \; \exists\textcolor{red}{!} P \; \forall x \; x \in P \leftrightarrow (x = a \lor x = b)
	\]
\end{proposition}

\begin{exercise}
	Dimostra la proposizione precedente.
\end{exercise}

\begin{proposition}
	Dato $a$, esiste ed \`e unico il \vocab{singoletto} $\{a\}$.
	\[ \forall a \; \exists ! S \; \forall x \; x \in S \leftrightarrow x = a
		\]
\end{proposition}

\begin{proof}
	Ponendo $b = a$ nella proposizione precedente, si ha che:
	\[ \forall a \; \exists ! S \; \forall x \; x \in S \leftrightarrow (x = a \lor x= a)
		\]
	ora $x = a \lor x = a$ equivale a $x = a$.
\end{proof}

\begin{notation}
\ \\
\vskip-4\baselineskip
\begin{align*}
{\drkblu P = \{a,b\}} &\;\;\Mydef\;\; \forall x \, x \in P \leftrightarrow (x = a
\lor x = b)\\
{\drkblu S = \{a\}} &\;\;\Mydef\;\; \forall x \, x \in S \leftrightarrow x = a
\end{align*}
\end{notation}

\begin{remark}
	$\{a,b\} = \{b,a\}$.
\end{remark}

\begin{proof}
	Segue dal fatto che $\lor$ \`e commutativo:
	\[ x \in \{a,b\} \leftrightarrow (x = a \lor x = b) \leftrightarrow (x = b \lor x = a) \leftrightarrow x \in \{b,a\}
		\]
	quindi per \hyperref[ax2]{estensionalit\`a} $\{a,b\} = \{b,a\}$.
\end{proof}

Il paio $\{a,b\}$ \`e, quindi, una coppia non ordinata. \`E possibile
codificare le coppie ordinate con questo trucco:

\begin{definition}[Coppia di Kuratowski]
\ \\
\vskip-3\baselineskip
	\[{\drkblu (a,b)} \;\;\Mydef\;\; \{\{a\},\{a,b\}\}
		\]
\end{definition}

\begin{proposition}
	La coppia di Kuratowski $(a,b)$ rappresenta la coppia ordinata di
$a$ e $b$, ossia:
	\[ (a,b) = (a^{\prime},b^{\prime}) \;\;\leftrightarrow\;\; (a = a^{\prime} \land b = b^{\prime})
		\]
\end{proposition}

\begin{proof}
	Detto $c = (a,b)$, vogliamo determinare univocamente $a$ e $b$.
Sostengo che valgono:
\begin{align*}
	x = a &\quad\leftrightarrow\quad \phantom{!}\forall {y \in c} \,(x \in y)
\tag{$\star$}\\
	x = b &\quad\leftrightarrow\quad \existsun {y \in c} \,(x \in y)
\tag{$\star\star$}
\end{align*}
	Formula ($\star$): la freccia $\rightarrow$ segue dalla definizione, mentre
$\leftarrow$ segue dal fatto che $\{a\}$ \`e un particolare elemento
di~$c$, e $x\in\{a\}$ implica $x=a$.

	Formula ($\star\star$): per la freccia $\rightarrow$, notiamo che
$y=\{a,b\}$ soddisfa $y\in c$ e $x=b\in y$. Bisogna quindi dimostrare che
se $y\in c$ e $x=b\in y$, allora $y=\{a,b\}$. L'unica altra possibilit\`a
per $y\in c$ \`e $y=\{a\}$, da cui $b\in\{a\}$, quindi $b=a$, quindi $y=\{a\}=\{a,b\}$.
La freccia $\leftarrow$ segue dal fatto che l'esistenza implica $y=\{a\}
\lor y=\{a,b\}$, da cui $x=a \lor
x=b$. Se $a=b$, quindi, non c'\`e nulla da dimostrare. Se $a\neq b$,
per\`o, $x=a$ non \`e accettabile, in quanto $\{a\}\neq\{a,b\}$ per
estensionalit\`a, e $a$ appartiene a entrambi contraddicendo l'unicit\`a.
\end{proof}

\begin{definition}[tuple ordinate]
\ 
\vskip-3\baselineskip
	\begin{align*}
	   {\drkblu (a,b,c)} &\;\;\Mydef\;\; ((a,b),c) \\
	   {\drkblu (a,b,c,d)} &\;\;\Mydef\;\; (((a,b),c),d) \\
	   \drkred (a_1,a_2,\ldots,a_n) &\drkred\;\;\Mydef\;\;
((a_1,a_2,\ldots,a_{n-1}),a_n)
	\end{align*}
\end{definition}

\begin{note}
	Quest'ultima definizione \textbf{non \`e una formula insiemistica
valida.} Si tratta, in realt\`a, di uno schema di definizioni: una per
ogni valore concreto di~$n$. Per ora, \textbf{non} siamo in grado di scrivere, per esempio,
	una formula insiemistica che dica ``Esiste un $n$ ed una $n$-tupla
$(a_1,\ldots,a_n)$ tale che $\dotsc$''. Per\`o, per ogni $n$ dato
concretamente, chess\`o 92, con un po' di pazienza,
possiamo scrivere una formula che dice $x = (a_1,a_2,a_3,\ldots,a_{92})$.
\end{note}

\begin{proposition}
\ 
\vskip-3\baselineskip
	\begin{align*}
(a,b,c) = (a',b',c') &\;\;\leftrightarrow\;\; a = a' \land b = b' \land c = c'\\
\drkred (a_1,\ldots,a_n) = (a_1',\ldots,a_n')
&\drkred\;\;\leftrightarrow\;\; a_1 = a_1' \land \ldots \land a_n = a_n'
\end{align*}
\end{proposition}

\begin{exercise}
	Dimostra la prima e convinciti che, dato un qualunque $n$ esplicito, potresti dimostrare la seconda.
\end{exercise}

\subsection{L'assioma dell'unione}
\vskip\baselineskip

\begin{axiom}[Assioma dell'unione]
	\label{ax5}
``Dato un insieme $A$ esiste un insieme $B$ i cui elementi sono gli
elementi degli elementi di $A$.''\\
\hbox to 2em{\hss}{\drkgrn o anche}\\
``Dato un insieme $A$ esiste l'unione degli elementi di $A$.''
	\[ \forall A \; \exists B \; \forall x \; x \in B \leftrightarrow \exists y \in A \; x \in y
		\]
\end{axiom}

\begin{proposition}
\ 
\vskip-3\baselineskip
	\[ \forall A \; \exists \textcolor{red}{!} B \; \forall x \; x \in B \leftrightarrow \exists y \in A \; x \in y
		\]
\end{proposition}

\begin{exercise}
	Dimostra la proposizione precedente.
\end{exercise}

\begin{notation}[Unione di un insieme]
\ 
\vskip-2\baselineskip
	\[ {\drkblu B = \union A} \;\;\Mydef\;\; \forall x \,( x \in B
\leftrightarrow \exists {y \in A} \; x \in y) \]
\end{notation}

\begin{exercise}
	Dimostra che l'assioma dell'unione segue da
	\[ \forall A \; \exists B \; (\forall y \in A \; \forall x \in y \; x \in B)
		\]
\end{exercise}

Combinando gli assiomi dell'unione e del paio possiamo definire $a \cup b$.

\begin{definition}
\ 
\vskip-3\baselineskip
	\[ {\drkblu a \cup b} \;\;\Mydef\;\; \union\{a,b\}
		\]
\end{definition}

\begin{proposition}
\ 
\vskip-3\baselineskip
	\[ x \in a \cup b \;\;\leftrightarrow\;\; (x \in a \lor x \in b)
		\]
\end{proposition}

\begin{proof}
	Dire che $x$ \`e un elemento di $a \cup b$ equivale a dire che $x$ \`e un elemento di un elemento di $\{a,b\}$, ossia
	che $x$ \`e un elemento di uno fra $a$ e $b$.
\end{proof}

Ora definiremo le intersezioni: \emph{riesci a vedere perch\'e, a
differenza delle unioni, non serve alcun nuovo assioma?}

\begin{definition}
	Sia $C$ una {\drkred classe} non vuota. {\drkgrn Nota che, in particolare, $C$ pu\`o
essere un insieme.}
	L'{\drkred insieme} $B$ \`e l'\vocab{intersezione} di $C$ se:
	\[ {\drkblu B = \bigcap C} \;\;\Mydef\;\; \forall x (x \in B \leftrightarrow
\forall {y \in C} (x \in y))
		\]
\end{definition}

\begin{proposition}
	Data una classe non vuota $C$, l'intersezione $\bigcap C$ esiste,
cio\`e \`e un insieme, ed \`e unica. In particolare, nel caso dell'intersezione di un insieme vale:
	\[ \forall A (A \ne \emptyset \rightarrow \exists ! B \; \forall x(x \in B \leftrightarrow \forall y \in A (x \in y)))
		\]
\end{proposition}

\begin{note}
	L'ipotesi $C \ne \emptyset$ \`e necessaria perch\'e altrimenti si avrebbe che $\bigcap \emptyset$ \`e la classe universale $V$, che non \`e un insieme.
\end{note}

\begin{proof}
	L'unicit\`a segue per \hyperref[ax2]{estensionalit\`a} al solito modo. Veniamo all'esistenza. Dal momento che $C$ non \`e vuota esiste $z \in C$,
	dunque per separazione possiamo costruire l'insieme seguente, che
dimostreremo essere $\bigcap C$:
	\[ B = \{x \in z \;|\; \forall {y \in C} (x \in y)\}
		\]
	Chiaramente $x \in B \implies \forall {y \in C} (x \in y)$.
D'altro canto, $\forall {y \in C} (x \in y)$ implica,
	in particolare, $x \in z$, per cui $x \in B$. Abbiamo cos\`i
verificato che $x \in B \leftrightarrow \forall {y \in C} (x \in y)$, ossia $B = \bigcap C$.
\end{proof}

\begin{notation}
\ 
\vskip-3\baselineskip
\begin{align*} {\drkblu a \cap b} &\;\;\Mydef\;\; \bigcap\{a,b\}\\
{\drkblu a\setminus b} &\;\;\Mydef\;\; \{x \in a | x \not\in b\}
\end{align*}
\end{notation}

\begin{proposition}
\ 
\vskip-3\baselineskip
\begin{align*}
	x \in a \cap b &\;\;\leftrightarrow\;\; (x \in a \land x \in b)\\
	x \in a \setminus b &\;\;\leftrightarrow\;\; (x \in a \land x \not\in b)
\end{align*}
\end{proposition}

\begin{exercise}
	Dimostrare la proposizione precedente (la seconda \`e semplicemente la definizione).
\end{exercise}

\begin{proposition}[Alcune propriet\`a delle operazioni $\cup$, $\cap$, $\setminus$]
\ 
\vskip-3\baselineskip
\begin{alignat*}{2}
\text{\drkblu{commutativit\`a:}}&\quad&   a \cup b &= b \cup a\\
	&& a \cap b &= b \cap a \\
\text{\drkblu{associativit\`a:}}&&   a \cup (b \cup c) &= (a \cup b) \cup c \drkblu\Mydef a \cup b \cup c \\
	&& a \cap (b \cap c) &= (a \cap b) \cap c \drkblu\Mydef a \cap b \cap c \\
\text{\drkblu{distributivit\`a:}}&&  a \cup (b \cap c) &= (a \cup b) \cap (a \cup c) \\
	&& a \cap (b \cup c) &= (a \cap b) \cup (a \cap c) \\
\text{\drkblu{leggi di De Morgan:}}&&   a \setminus (b \cup c) &= (a \setminus b) \cap (a \setminus c) \\
	&& a \setminus (b \cap c) &= (a \setminus b) \cup (a \setminus c)
\end{alignat*}
\end{proposition}

\begin{proof}
	Tutte queste propriet\`a si deducono immediatamente dalle corrispondenti propriet\`a dei connettivi logici, le quali, a loro volta, si vedono con le tabelle di verit\`a. Per esempio, dimostriamo 
	la prima delle leggi di De Morgan.
\begin{alignat*}{2}
		x \in a \setminus (b \cup c) &\quad\iff\quad & \mathrlap{x \in a \land x \not\in (b \cup c)}\phantom{x \in a \land x \not\in b} &\\
		&\quad\iff & \mathrlap{x \in a \land \neg(x \in b \lor x \in c)}\phantom{x \in a \land x \not\in b} &\\
		&\quad\iff & x \in a \land x \not\in b &\land x \not\in c\\
		&\quad\iff & x \in a \land x \not\in b &\land x \in a \land x \not\in c\\
		&\quad\iff & x \in (a \setminus b)\;\; &\land\;\;\; x \in (a \setminus c)\\
		&\quad\iff & x \in (a \setminus &b) \cap (a \setminus c)
\end{alignat*}
\end{proof}

Ora possiamo costruire insiemi finiti elencandone gli elementi, come si fa di solito, con la notazione $\{\ldots\}$.

\begin{notation}
\ 
\vskip-3\baselineskip
	\begin{align*}
	   {\drkblu \{a,b,c\}} &\;\;\Mydef\;\; \{a\} \cup \{b\} \cup \{c\} \\
	   {\drkblu \{a,b,c,d\}} &\;\;\Mydef\;\; \{a\} \cup \{b\} \cup \{c\} \cup \{d\} \\
	   \drkred \{a_1,\ldots,a_n\} &\drkred\;\;\Mydef\;\; \{a_1\} \cup \ldots \cup \{a_n\}
	\end{align*}
\end{notation}

\begin{proposition}
\ 
\vskip-3\baselineskip
	\begin{align*}
		x \in \{a,b,c\} & \;\;\leftrightarrow\;\; (x = a \lor x = b \lor x = c) \\
	    \drkred x \in \{a_1,\ldots,a_n\} &\drkred \;\;\leftrightarrow\;\; (x = a_1 \lor \ldots \lor x = a_n) 
	\end{align*}
\end{proposition}

\begin{exercise}
	Dimostrare la proposizione precedente.
\end{exercise}

\subsection{Assioma delle parti e prodotto cartesiano}
Lo scopo della teoria che stiamo sviluppando \`e di permettere, mediante
opporune codifiche, tutte le costruzioni della matematica ordinaria. Per
esempio, abbiamo visto, la coppia di Kuratowski permette di codificare le
coppie ordinate per mezzo di insiemi opportuni. Avendo a disposizione
coppie ordinate, possiamo dire, per esempio, che un certo insieme \`e una
funzione da~$A$ a~$B$. Per\`o ancora ci manca un modo per costruire
l'insieme di \textit{tutte} le funzioni da~$A$ a~$B$. Lo otterremo
sfruttando l'ingrediante di questa sezione: l'\vocab{insieme delle parti,}
detto anche \vocab{insieme potenza.}

\begin{axiom}
	[Assioma delle parti]
	\label{ax6}
	``Dato un insieme $A$ esiste un insieme $B$ i cui elementi sono i
sottoinsiemi di $A$.''
	\[ \forall A \; \exists B \; \forall x \; x \in B \leftrightarrow x \subseteq A 
		\]
\end{axiom}

\begin{proposition}
\ 
\vskip-3\baselineskip
	\[\forall A \; \existrun B \; \forall x \; x \in B \leftrightarrow x \subseteq A 
		\]
\end{proposition}
\begin{proof}
A questo punto non vale neanche pi\`u la pena di fare l'esercizio.
\end{proof}

\begin{notation}
\ 
\vskip-3\baselineskip
	\[ {\drkblu B = \ps(A)} \;\;\Mydef\;\; \forall x \; x \in B \leftrightarrow x \subseteq A
		\]
L'insieme $\ps(A)$ \`e detto \vocab{insieme delle parti} o \vocab{insieme
potenza} di~$A$.
\end{notation}

\begin{proposition}
	Dati $A$ e $B$  esiste un unico insieme \vocab{$A \times B$} tale che:
	\[ \forall z (z \in A \times B) \;\;\leftrightarrow\;\; \exists {x
\in A} \, \exists {y \in B} \, z = (x,y)
		\]
\drkgrn Ossia, informalmente
\[ A\times B \;\;=\;\; \{\,(x,y)\,|\,x\in A \land y\in B\,\} \]
\end{proposition}

\begin{proof}
	L'unicit\`a \`e conseguenza immediata della definizione e
dell'\hyperref[ax2]{assioma di estensionalit\`a.} Per l'esistenza,
definiamo per \hyperref[ax3]{separazione:}
	\[ A \times B \;\;\Mydef\;\; \{\,z \in \ps(\ps(A \cup B))\, |\, \exists {x
\in A} \exists {y \in B}  z = (x,y)\,\}
		\]
Dobbiamo dimostrare che ogni coppia $(x,y)$, con $x\in A$ e~$y\in B$,
appartiene a questo insieme. Basta dimostrare che tutte queste coppie
appartengono a~$\ps(\ps(A \cup B))$.
\begin{alignat*}{2}
		x \in A \land y \in B &\quad\implies\quad& \mathrlap{\{x\}\subseteq
A\cup B}\phantom{\{x\} \in \ps(A\cup B)} &\;\;\land\;\; \{x,y\} \subseteq A \cup B \\
		& \quad\implies\quad& \{x\} \in \ps(A\cup B)
&\;\;\land\;\; \{x,y\} \in \ps(A \cup B) \\
		& \quad\implies\quad& (x,y) = \{\{x\},\,&\{x,y\}\} \subseteq \ps(A \cup B)\\
		& \quad\implies\quad& &\phantom{\{x,y\}\}}\mathllap{(x,y)}
\in \ps(\ps(A \cup B))
\end{alignat*}
\vskip-\baselineskip\qedhere
\end{proof}

\begin{note}
	Avremmo potuto costruire $A \times B$ usando, anzich\'e l'assioma delle parti, l'assioma del rimpiazzamento, che vedremo pi\`u avanti.
\end{note}

\subsection{Relazioni di equivalenza e di ordine, funzioni}
Ora rivedremo alcuni concetti ben noti dai primi corsi del primo anno (\emph{o dalla scuola superiore?}). Lo facciamo molto rapidamente, essenzialmente per completezza, e per fissare le notazioni.

\begin{definition}
Una \vocab{relazione binaria} fra $A$ e $B$ \`e un sottoinsieme di $A \times B$.
\end{definition}

\begin{notation}
	Data una relazione binaria $\rel \subseteq A \times B$, definiamo l'abbreviazione:
	\[ {\drkblu a\,\rel\,b} \;\;\Mydef\;\; (a,b) \in \rel
		\]
{\drkgrn Per esempio scriviamo $a < b$ per indicare che $(a,b) \in\, <$.}
\end{notation}

\begin{notation}
Per brevit\`a scriveremo $\forall x_1,x_2\dotsc x_n$ al posto di $\forall x_1
\forall x_2 \dotsc \forall x_n$, e similmente con~$\oldexists$. Scriveremo
$\forall x_1,x_2\dotsc x_n\in Y$ al posto di $\forall x_1\in Y\;\forall x_2\in Y
\;\dotsc \forall x_n\in Y$, e similmente con~$\oldexists\dotsc\in Y$.
\end{notation}

\begin{definition}
	La relazione $\sim \,\subseteq A \times A$ \`e una \vocab{relazione di equivalenza} se \`e:
\begin{alignat*}{2}
&\text{\drkblu riflessiva:} && \;\;\forall {x \in A} \; x \sim x\\
&\text{\drkblu simmetrica:} && \;\;\forall {x,y \in A} \; x\sim y \leftrightarrow y \sim x\\
&\text{\drkblu transitiva:} && \;\;\forall {x,y,z \in A} \; (x \sim y \land y \sim z) \rightarrow x \sim z
\end{alignat*}
\end{definition}

\begin{definition}
	La relazione $\leq \, \in A \times A$ \`e una \vocab{relazione di ordine largo} se \`e:
\begin{alignat*}{2}
&\text{\drkblu riflessiva:} && \;\;\forall {x \in A} \; x \leq x\\
&\text{\drkblu antisimmetrica:} && \;\;\forall {x,y \in A} \; (x \leq y \land y \leq x) \rightarrow x = y\\
&\text{\drkblu transitiva:} && \;\;\forall {x,y,z \in A} \; (x \leq y \land y \leq z) \rightarrow x \leq z
\end{alignat*}
\end{definition}

\begin{definition}
	La relazione $< \, \in A \times A$ \`e una \vocab{relazione di ordine stretto} se \`e:
\begin{alignat*}{2}
&\text{\drkblu irriflessiva:} && \;\;\forall {x \in A} \; \neg(x < x)\\
&\text{\drkblu transitiva:} && \;\;\forall {x,y,z \in A} \; (x < y \land y < z) \rightarrow x < z
\end{alignat*}
\end{definition}

\begin{exercise}
	Dimostra che una relazione di ordine stretto $<$ su $A$ \`e asimmetrica:
	\[ \forall x,y \in A \; x < y \rightarrow \neg (y < x)
		\]
\end{exercise}

\begin{proposition}\label{precedente}
	Data una relazione di ordine stretto $<$ su $A$, la relazione:
	\[ \leq \;\;=\;\; \{(x,y) \in A \times A | x < y \lor x = y\} \]
	\`e una relazione di ordine largo. Viceversa, se $\leq$ \`e una
relazione di ordine largo, la relazione seguente \`e di ordine stretto:
	\[ < \;\;=\;\; \{(x,y) \in A \times A | x \leq y \land x \ne y\}
		\]
	Inoltre, la corrispondenza stabilita in questo modo \`e biunivoca.
\end{proposition}

\begin{note}
La notazione
\[\{(x,y)\in A \times A | \dotsc\}\]
nella proposizione precendete non \`e stata definita. Formalmente, avrei
dovuto fissare $B=A\times A$ e scrivere
\[\{z \in B | \exists x,y \in A \, z=(x,y)\land \dotsc\}.\]
Abusi di questo genere continueranno, e saranno presto passati sotto
silenzio.
\end{note}

\begin{proof}[Dimostrazione di~\ref{precedente}]
	Definiamo la \vocab{diagonale}
	\[ {\drkblu \Delta_A} \;\;\Mydef\;\; \{(x,y) \in A \times A | x = y\}
		\]
	\`E facile verificare che, se $<$ \`e una relazione di ordine
stretto, allora $< \cap \,\Delta_A = \emptyset$ e $< \cup \,\Delta_A$ \`e
la relazione di ordine largo corrispondente.
	Viceversa, se $\leq$ \`e una relazione di ordine largo, allora $\Delta_A \subseteq \, \leq$ e $\leq \setminus \Delta_A$ \`e la relazione di ordine stretto corrispondente.
\end{proof}

\begin{notation}
	Fissata una relazione di ordine largo $\textcolor{red}{\leq}$ su
$A$, ci sentiremo liberi di usare la corrispondente relazione di ordine
stretto $\textcolor{red}{<}$, e viceversa, fidando che la scelta del simbolo sia indizio sufficiente.
Parleremo quindi semplicemente di \vocab{relazione di ordine} come se non
avessimo paura di essere fraintesi.
	Inoltre scriveremo $x > y$ per $y < x$ e $x \geq y$ per $y \leq x$.
\end{notation}

\begin{definition}[Ordini totali]
	Una \vocab{relazione di ordine totale} su $A$ \`e una relazione di ordine $\leq$ tale che:
	\[ \forall x,y \in A \, (x \leq y) \lor (x = y) \lor (y \leq x)
		\]
\end{definition}

\begin{exercise}
	Formula la definizione precedente per ordini stretti.
\end{exercise}

\begin{definition}[Restrizione di una relazione]
	Data una relazione $\rel \subseteq A \times B$, e dati $A'
\subseteq A$, $B' \subseteq B$, possiamo definire la \vocab{restrizione}
di $\rel$ ad $A'\times B'$:
	\[ {\drkblu \rel_{|A' \times B'}} \;\;\Mydef\;\; \rel \cap (A' \times B')
		\]
\end{definition}

\begin{exercise}
	Data $\rel$ relazione di equivalenza/ordine su $A$ e $A' \subseteq A$, dimostra che $\rel_{|A' \times A'}$ \`e una relazione di equivalenza/ordine su $A'$.
\end{exercise}

\begin{definition}
	Data una relazione $\rel \subseteq A \times B$, definiamo:
	\[  \begin{split}
		\Dom(\rel) &\;\;\Mydef\;\; \{x \in A | \exists {y \in B} \;
x\,\rel\, y\} \qquad \text{\vocab{dominio} di $\rel$} \\
	    \Imm(\rel) &\;\;\Mydef\;\; \{y \in B | \exists {x \in A} \; x\,
\rel\, y \} \qquad \text{\vocab{immagine} di $\rel$}
	\end{split}
			\]
\end{definition}

\begin{definition}
	Chiamiamo \vocab{funzione $f:A \rightarrow B$} una relazione $f \subseteq A \times B$ tale che:
	\[ \forall {x \in A} \; \existsun {y \in B} \; (x,y) \in f
		\]
	{\drkgrn Intuitivamente $f$ \`e l'insieme delle coppie $(x,f(x))$
per $x \in A$.}
\end{definition}

\begin{notation}
	Data una funzione $f$, scriviamo
	\[ {\drkblu y = f\drkred (\drkblu x\drkred )} \;\;\Mydef\;\; (x,y) \in f
		\]
	Dato $S \subseteq \Dom(f)$, scriviamo
	\[ {\drkblu f\drkred [\drkblu S\drkred ]} \;\;\Mydef\;\; \{y \in
\Imm(f)| \exists {x \in S }\; y = f(x) \} \drkgrn \;\;=\;\; \{f(x) | x \in S\}
		\]
\end{notation}

\begin{definition}
	Una funzione $f: A \rightarrow B$ \`e:
	\[ \begin{split}
		\text{\vocab{iniettiva} se:}\; & \forall y \in \Imm(f)\; \exists! \, x \in \Dom(f) \; f(x) = y\\
		\text{\vocab{surgettiva} se:}\; & B = \Imm(f)\; \text{ossia $\forall y \in B \; \exists x \in A \; f(x) = y$.}\\
		\text{\vocab{bigettiva} se:}\; &\text{\`e sia iniettiva sia surgettiva.}
	\end{split}
		\]
\end{definition}

\begin{definition}[Funzione inversa]
	Data $f$ iniettiva:
	\[ f^{-1} \;\;\Mydef\;\; \{(y,x) \in B \times A | f(x) = y\}
\;\;\subseteq\;\; B \times A
		\]
\end{definition}

\begin{remark}
	Data $f$ iniettiva, $f^{-1}$ \`e una funzione $\Imm(f) \to
\Dom(f)$ iniettiva.
	In particolare, se $f \colon A \to B$ \`e bigettiva, allora
$f^{-1}$ \`e bigettiva $B\to A$.
\end{remark}

\begin{definition}
	Data $f: A \rightarrow B$ e $A' \subseteq A$ definiamo la
\vocab{restrizione di $f$ ad $A'$:}
	\[ {\drkblu f_{|A'}} \;\;\Mydef\;\; \{(x,y) \in A' \times B | f(x) = y\}
		\]
\end{definition}

\begin{definition}
	Date $g : A \rightarrow B$ e $f : B \rightarrow C$, definiamo la
loro \vocab{composizione $f\circ g\colon A\to C$:}
	\[ {\drkblu f \circ g} \;\;\Mydef\;\; \{(x,z) \in A \times C | z =
f(g(x))\} \]
\end{definition}

\begin{notation}
	Indichiamo con $\id_A$ la \vocab{funzione identit\`a} su $A$:
	\[ {\drkblu \id_A} \;\;\Mydef\;\; \{(x,y) \in A \times A | x = y\}
\;\;=\;\; \Delta_A
		\]
\end{notation}

\begin{remark}
	Data $f : A \rightarrow B$ bigettiva e $g : B \rightarrow A$ \`e equivalente scrivere:
	\[ g = f^{-1} \qquad g \circ f = \id_A \qquad f \circ g = \id_B
		\]
\end{remark}

\begin{exercise}
	Data $f\colon A \to B$ e $g: B \rightarrow C$, sotto quali condizioni $g \circ f$ \`e iniettiva, surgettiva, bigettiva?
\end{exercise}

\begin{exercise}
	\label{3.73}
	Data una relazione di equivalenza $\sim$ su $A$, dimostra che esiste un insieme $\faktor{A}{\sim}$ ed una funzione surgettiva $i_\sim$ da $A$ a $\faktor{A}{\sim}$
	tale che:
	\[ \forall x,y \in A \; x \sim y \leftrightarrow i_\sim(x) = i_\sim(y)
		\]
\end{exercise}


\begin{exercise}
	Data una relazione di equivalenza $\sim$ su $A$ e $f : A \rightarrow B$, affinch\'e esista la funzione $\widetilde{f}: \faktor{A}{\sim} \rightarrow B$ tale che $f = \widetilde{f} \circ i_\sim$,
	\`e necessario e sufficiente che $\forall x,y \in A \; x \sim y \rightarrow f(x) = f(y)$.
\end{exercise}
