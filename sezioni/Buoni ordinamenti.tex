\section{I buoni ordini}
Il nostro prossimo obiettivo \`e definire e studiare la classe dei \vocab{numeri ordinali}. Questa pu\`o essere pensata come la pi\`u vasta classe
-- dotata di un ordine totale definito per mezzo di una formula -- su cui sia corretto ragionare per induzione forte. Conteremo, quindi, sugli ordinali per formulare 
l'induzione e la ricorsione transfinita, procedimenti che superano la forza dimostrativa dell'induzione e della ricorsione aritmetica -- per esempio permettendo di ottenere 
il teorema di Cantor-Lebesgue sugli insiemi di unicit\`a.

Siccome l'induzione forte equivale al principio del minimo, studieremo i buoni ordini. In questa sezione, dimostreremo il risultato seguente.

\begin{theorem}
	Siano $(A,<_A)$ e $(B,<_B)$ {\drkgrn -- scriviamo semplicemente $(A,<)$
e~$(B,<)$, per comodit\`a --} insiemi bene ordinati. Allora vale una e una
sola delle seguenti:
	\begin{itemize}
		\item $(A,<)$ \`e isomorfo a un segmento iniziale proprio di $(B,<)$
		\item $(A,<)$ e $(B,<)$ sono isomorfi
		\item $(B,<)$ \`e isomorfo a un segmento iniziale di $(A,<)$
	\end{itemize}

Quindi, definendo
\[ (A,<) \prec (B,<) \;\;\Mydef\;\; \exists C \; \text{$C$ \`e segmento
iniziale proprio di $(B,<)$ e $(A,<) \sim C$}
	\]
$\prec$ soddisfa le propriet\`a formali di un ordine totale fra le classi di isomorfismo di buoni ordini.
\end{theorem}

\begin{remark}
L'ordine largo associato a $\prec$ \`e:
\[ (A,<) \preceq (B,<) \;\;\Mydef\;\; (A,<) \prec (B,<) \lor (A,<) \sim (B,<)
	\]
ossia ``$(A,<)$ \`e isomorfo a un segmento iniziale di $(B,<)$''.
\end{remark}

Richiamiamo alcune definizioni fondamentali.

\begin{definition}[Buon ordine]
	$(A,<)$ \`e un \vocab{buon ordine} se ogni $B \subseteq A$ non vuoto ha un minimo elemento.
\end{definition}

\begin{definition}[Segmento inziale]
	Dato un ordine totale $(A,<)$, $B \subseteq A$ \`e un
\vocab{segmento inziale} di~$A$ se
\[\forall {b \in B} \; \forall {x \in A} \; x < b \rightarrow x \in B\]
\end{definition}

\begin{definition}[Segmenti iniziali propri e principali]
	Il segmento iniziale $B$ \`e \vocab{proprio} se $B \ne A$. Il segmento iniziale $B$ \`e \vocab{principale} se \`e della forma:
	\[ B \;=\; A_a \;\Mydef\; \{x \in A | x < a\}
		\]
	per qualche $a \in A$, e, in questo caso, si dice  che \`e il \vocab{segmento iniziale principale determinato da $a$}.
\end{definition}

Un segmento iniziale principale, $A_a$, \`e automaticamente proprio,
perch\'e $a \not \in A_a$. Nel caso dei buoni ordini questa \`e una doppia
implicazione:

\begin{proposition}
	Ogni segmento iniziale proprio di un buon ordine \`e principale.
\end{proposition}

\begin{proof}
	Sia $(A,<)$ ben ordinato e $I \subsetneq A$ un segmento iniziale
proprio. Consideriamo $a = \min (A \setminus I)$. Allora $I = A_a$.

	{\drkgrn Verifiche. $x\in A_a \Rightarrow x < a \Rightarrow
x\not\in A\setminus I \Rightarrow x\in I$.\\
Viceversa, se $x\in I$, non pu\`o valere $a\le x$, perch\'e $a\not\in I$,
quindi $x<a$, cio\`e $x\in A_a$.}
\end{proof}

\begin{exercise}
	Dimostra che la proposizione precedente caratterizza i buoni ordini. Pi\`u precisamente, dato un ordine totale $(A,<)$, se ogni segmento iniziale proprio di $A$
	\`e principale, allora $A$ \`e bene ordinato da $<$.
\end{exercise}

\begin{exercise}
Perch\'e non \`e vero che, in~$\ZZ$, ogni segmento iniziale proprio \`e
principale?
\end{exercise}

\begin{lemma}
	Sia $(A,<)$ un buon ordine e $f \colon A \to A$ una funzione
strettamente crescente, allora $\forall {x \in A} \; x \leq f(x)$.
\end{lemma}

\begin{proof}
	Per assurdo, assumiamo la negazione della tesi. Quindi l'insieme $B = \{x \in A | f(x) < x\}$ non \`e vuoto. Sia $k = \min B$.
	Allora $f(k) < k$ per la definizione di~$B$. Applicando $f$ ad
ambo i membri, siccome $f$ \`e
crescente, $f(f(k))<f(k)$, ossia $f(k) \in B$. Ricordando che $f(k)<k$, per\`o,
questo contraddice la minimalit\`a di~$k$.
\end{proof}

\begin{corollary}\label{cor-bonanza}
\ 
\vskip-2.5\baselineskip
\ 
	\begin{enumerate}[(1)]
		\item Un buon ordine non \`e isomorfo a un suo segmento iniziale proprio.
		\item L'identit\`a \`e il solo isomorfismo fra un buon ordine e se stesso.
		\item Se $(A,<)$ e $(B,<)$ sono buoni ordini isomorfi allora esiste un unico isomorfismo fra di essi.
	\end{enumerate}
\end{corollary}

\begin{proof}\ 

\vskip-2\baselineskip
	\begin{enumerate}[(1)]
		\item Sia $f \colon A \to A_a$ un isomorfismo.
		In particolare $f$ \`e crescente. Tuttavia $f(a) <
a\;\lightning$
		\item Sia $f \colon A \to A$ un automorfismo del buon
ordine $(A,<)$. Allora sia $f$ sia $f^{-1}$ sono crescenti. Quindi, per
ogni $x\in A$, valgono
$x\le f(x)$ e $x\le f^{-1}(x)$. Dalla seconda, per\`o, segue $f(x)\le x$,
che, combinata con la prima, implica $f(x)=x$.
		\item Se $f \colon A \to B$ e $g \colon A \to B$ fossero
due isomorfismi diversi, allora $g^{-1} \circ f \colon A \to A$ sarebbe un
automorfismo di $A$ diverso dall'identit\`a.
	\end{enumerate}
\end{proof}

\begin{remark}
	Siano $(A,<)$, $(B,<)$, $(C,<)$ buoni ordini, allora:
	\[ (A,<) \preceq (B,<) \;\land\; (B,<) \preceq (C,<)
\;\;\rightarrow\;\; (A,<) \preceq (C,<)
		\]
\end{remark}

\begin{proof}
	Siano $f\colon A \to B$ e $g \colon B \to C$ isomorfismi fra $A$ e un segmento iniziale di $B$ e fra $B$ e un segmento iniziale di $C$ rispettivamente.
	Dimostriamo che $g \circ f \colon A \to C$ \`e un isomorfismo fra $A$ e un segmento iniziale di $C$. Naturalmente $g \circ f$ \`e crescente in quanto composizione di funzioni crescenti, dunque occorre solo verificare che $g \circ f[A]$ \`e un segmento iniziale di $C$.

{\drkgrn Verifica. Sia $g(f(a))$ un qualunque elemento di $g \circ f[A]$, e sia $x \in C$ tale che $x < g(f(a))$, dobbiamo verificare che $x \in g \circ f[A]$.
	Siccome $g(f(a)) \in g[B]$ e, per ipotesi, $g[B]$ \`e un segmento
iniziale di $C$, otteniamo $x \in g[B]$. Possiamo allora scrivere $x =
g(y)$. Ora, siccome $g$ \`e un isomorfismo, da $x = g(y) < g(f(a))$
deduciamo $y < f(a)$, inoltre, per ipotesi, $f[A]$
	\`e un segmento iniziale di $B$, pertanto $y \in f[A]$, ovvero $y = f(z)$ per $z \in A$. Abbiamo quindi $x = g(f(z)) \in g \circ f[A]$.}
\end{proof}

\begin{remark}
	Siano $(A,<)$ e $(B,<)$ buoni ordini, allora:
	\[ (A,<) \preceq (B,<) \;\land\; (B,<) \preceq (A,<)
\;\;\rightarrow\;\; (A,<) \sim (B,<)
		\]
\end{remark}

\begin{proof}
	Siano $f \colon A \to B$ e $g \colon B \to A$ isomorfismi fra $A$ e un segmento iniziale di $B$ e fra $B$ e un segmento iniziale di $A$.
	Ricordando la dimostrazione dell'osservazione precedente, $g \circ
f$ \`e un isomorfismo fra $A$ e un segmento iniziale $g \circ f[A]$ di
$A$, ma per il corollario~\ref{cor-bonanza}~(1)
	$g \circ f[A]$ non pu\`o essere un segmento iniziale proprio di
$A$, quindi $g\circ f[A]=A$.
	Per il corollario~\ref{cor-bonanza}~(2), $g \circ f = \id_A$.
Ragionando simmetricamente si ottiene $f \circ g = \id_B$.
\end{proof}

Possiamo finalmente passare alla dimostrazione del teorema.

\begin{theorem}
	Siano $(A,<)$ e $(B,<)$ insiemi ben ordinati, allora vale una e una sola delle seguenti:
	\[ (A,<) \prec (B,<) \qquad (A,<) \sim (B,<) \qquad (B,<) \prec (A,<)
		\]
\end{theorem}

\begin{wrapfigure}[16]{l}{.3\textwidth}
	\includegraphics[width = .3\textwidth]{immagini/ordine_totale_buoni_ordini.png}
\end{wrapfigure}

{\drkgrn Idea: Il corollario ci dice che vale al pi\`u una delle alternative, quindi la difficolt\`a risiede nel dimostrare che una si verifica. Molto vagamente potremmo ragionare cos\`i.
Identifichiamo, progressivamente, segmenti iniziali sempre pi\`u lunghi di
$A$ e $B$. All'inizio identifichiamo il minimo di $A$ con il minimo di
$B$, poi il secondo elemento di $A$ con il secondo elemento di $B$, \etc. Fatti $\omega$
passi avremo identificato un segmento iniziale di $A$, diciamo $A_x$, isomorfo a $\omega$, con un $B_y$, anch'esso ovviamente isomorfo a $\omega$. Bene: continuiamo identificando $x$ con $y$. Quando potrebbe bloccarsi il procedimento?
Solo se, ad un certo punto, abbiamo identificato interamente uno dei due
insiemi, con un segmento iniziale dell'altro -- perch\'e altrimenti, abbiamo identificato due segmenti iniziali $A_x$ e $B_y$ e possiamo continuare attaccando $x$ a $y$. \`E 
come la chiusura di una cerniera lampo: ad ogni istante c'\`e un prossimo dente.

{\drkred Questo discorso, per\`o, non \`e una dimostrazione.} Se vogliamo,
sarebbe un tentativo di costruire l'isomorfismo cercato per
\vocab{ricorsione transfinita.} Il guaio \`e che i numeri che permetterebbero di numerare i passaggi della costruzione,
gli \vocab{ordinali,} sono appunto l'oggetto che stiamo tentando di costruire.}

\begin{proof}
	Per il corollario visto prima, si pu\`o verificare al pi\`u una
delle tre condizioni. Dimostriamo dunque che se ne verifica almeno una.
Consideriamo l'insieme $f$ cos\`i definito
	\[ f \;\;=\;\; \{(a,b) \in A \times B | A_a \sim B_b\}
		\]
	Vogliamo dimostrare che $f$ \`e una funzione, che \`e crescente,
che $\Dom(f)$ \`e un segmento iniziale di $A$, e che $\Imm(f)$ \`e un
segmento iniziale di $B$. Quindi $f$ \`e un isomorfismo fra un segmento iniziale di $A$ e uno di $B$.
	Infine dimostreremo che $\Dom(f) = A$ o $\Imm(f) = B$, e questo
conclude la dimostrazione.

{\drkblu $f$ \`e una funzione.} Supponiamo per assurdo $(a,b) \in f$ e
$(a,b') \in f$ con $b \ne b'$. Per la costruzione di~$f$ abbiamo:
		\[ B_b \sim A_a \sim B_{b'}
			\]
D'altro canto, senza perdita di generalit\`a, possiamo supporre $b < b'$, allora $B_b$ \`e un
segmento iniziale proprio di~$B_{b'}$, contraddicendo il fatto
che~$B_b\sim B_{b'}$.

{\drkblu $f$ \`e crescente.} Dati $a,a' \in A$, con $a<a'$, dobbiamo dimostrare $f(a) < f(a')$. Supponiamo, per assurdo $f(a') \leq f(a)$, abbiamo allora:
		\[ A_{a'} \sim B_{f(a')} \preceq B_{f(a)} \sim A_a \prec
A_{a'}
			\]
dove i due isomorfismi seguono dalla costruzione di~$f$, mentre le due
disuguaglianze seguono dalle inclusioni $B_{f(a')} \subseteq B_{f(a)}$ e
$\sim A_a\subsetneq  A_{a'}$.
Abbiamo ottenuto $A_{a'}\prec A_{a'}\;\lightning$

{\drkblu $\Dom(f)$ \`e s.i.\ di $A$.} Sia $a \in \Dom(f)$ e $a' < a$,
vogliamo dimostrare che $a' \in \Dom(f)$. Osserviamo che $A_{a'}$ \`e un
segmento iniziale proprio di $A_a$. L'isomorfismo che attesta $A_a\sim
B_{f(a)}$, manda quindi $A_{a'}$ in un segmento iniziale proprio di
$B_{f(a)}$. Questo \`e, in particolare, un segmento iniziale proprio
$B_{b'}$ di~$B$. Quindi $(a',b')\in f$.

{\drkblu $\Imm(f)$ \`e s.i. di $B$.} Dimostrazione simmetrica alla precedente.

{\drkblu $\Dom(f) = A$ o $\Imm(f) = B$.} Se cos\`i non fosse, avremmo $\Dom(f) = A_a$ e $\Imm(f) = B_b$
		per opportuni $a \in A$ e $b \in B$. Siccome $f$ \`e
crescente $A_a = \Dom(f) \sim \Imm(f) = B_b$.  Abbiamo quindi $A_a \sim
B_b$, per cui $(a,b)\in f\;\lightning$
\end{proof}

\begin{exercise}
	Sia $(A,<)$ un buon ordine e sia $B \subsetneq A$. Dimostra che $B \preceq A$, ma non necessariamente $B \prec A$.
\end{exercise}

\begin{exercise}
	Sia $(A,<)$ un ordine totale con $A = \bigcup S$. Supponiamo che:
	\begin{enumerate}[1.]
		\item ogni $X \in S$ \`e bene ordinato dalla restrizione
di~$<$ a~$X$
		\item per ogni $X,Y \in S$, o $X$ \`e segmento iniziale di $Y$ o $Y$ \`e segmento iniziale di $X$
	\end{enumerate}
	Dimostra che $(A,<)$ \`e un buon ordine. Esibisci inoltre un controesempio eliminando la condizione 2 dalle ipotesi.
\end{exercise}

\subsection{Operazioni aritmetiche fra buoni ordini}
Per ora, non abbiamo visto molti esempi di buoni ordini. Le operazione definite in questa sezione forniscono una prima sorgente di esempi concreti.
Nel seguito del corso, vedremo buoni ordini assai pi\`u vasti di quelli ottenibili con queste operazioni.

\begin{definition}[Somma di ordini totali]
	Dati $(A,<_A)$ e $(B,<_B)$ ordini totali. Definiamo
	\[ {\drkblu (A,<_A) + (B,<_B)} \;\;\Mydef\;\; (A \sqcup B, <_+)
		\]
	dove, ricordiamo che $A \sqcup B = (A \times\{0\}) \cup (B \times \{1\})$, e $<_+$ \`e definito da:
	\begin{align*}
		{\drkblu (x,y) <_+ (x',y')} \;\;\Mydef&\phantom{\lor}\; (y = 0 \land y' = 1) \\
						  &{\lor}\; (y = 0 \land y' = 0 \land x <_A x') \\
						  &{\lor}\; (y = 1 \land y' = 1 \land x <_B x')
	\end{align*}
\end{definition}

{\drkgrn Idea: $(A,<_A) + (B,<_B)$ si ottiene attaccando $(B,<_B)$ in coda a $(A,<_A)$}.

\begin{figure}[H]
	\centering
	\includegraphics[width = .7\textwidth]{immagini/somma_ordini_totali.png}
\end{figure}

Ripetiamo, per completezza, la definizione di prodotto lessicografico.

\begin{definition}[Prodotto di lessicografico]
	Siano $(A,<_A)$ e $(B,<_B)$ ordini totali. Definiamo
	\[ {\drkblu (A,<_A) \cdot (B,<_B)} \;\;\Mydef\;\; (A \times B, <_\times)
		\]
	dove $<_\times$ \`e definito da:
	\[ {\drkblu (x,y) <_\times (x',y')} \;\;\Mydef\;\; (y <_B y') \lor (y = y' \land x <_A x')
		\]
\end{definition}

{\drkgrn Idea: confronto prima la seconda componente; a parit\`a conta la
prima. Quindi $(A,<_A) \cdot (B,<_B)$ pu\`o essere pensato come tante
copie di~$(A,<_A)$ messe, fra loro, nello stesso ordine degli elementi
di~$(B,<_B)$: in breve, $(A,<_A)$ ripetuto $(B,<_B)$~volte.}

\begin{figure}[H]
	\centering
	\includegraphics[width = \textwidth]{immagini/ordine_lessicografico.png}
\end{figure}

Per definire l'esponenziale ci serve la nozione di supporto.

\begin{definition}
	Dato un buon ordine $(B,<)$ e $f\colon A \to B$, il \vocab{supporto} di $f$ \`e:
	\[ {\drkblu\supp_B(f)} \;\;\Mydef\;\; \{\:x \in A\: |\: f(x) \ne \min_B\:\}
		\]
	Scriveremo semplicemente $\drkblu\supp(f)$ quando \`e chiaro cosa sia $B$.
\end{definition}

\begin{definition}[Potenza di ordini totali]
	Dati $(A,<_A)$ e $(B,<_B)$ ordini totali, definiamo
	\[ {\drkblu (A,<_A)^{(B,<_B)}} \;\;\Mydef\;\; \Big(\;\big\oldparen
\underbrace{f \in {}^{B}A \: \big| \: |\supp(f)\,| <
\aleph_0}_{\text{funzioni $B\to A$ a supporto finito}}\big\}\;,\;<_{\exp}\;\Big)
		\]
	dove $<_{\exp}$ \`e definito da:
	\begin{align*} {\drkblu f <_{\exp} g} \;\;\Mydef\;\; f \ne
g\;\;\text{e}\;\;&\text{detto $m=\max\{x\in B|f(x)\neq g(x)\}$}\\
&\text{vale $f(m) < g(m)$}
\end{align*}
\end{definition}

{\drkgrn L'idea \`e che una funzione $B \rightarrow A$ pu\`o essere vista
come una specie di tupla con tante componenti quanti sono gli elementi di
$B$.
Ordinare due tuple lessicograficamente significa confrontare
la componente corrispondente all'elemento di $B$ pi\`u
grande su cui le funzioni siano diverse.}

\begin{exercise}
	Verifica che $(\omega,<)^{(\omega,<)} \sim (\NN[x],\prec)$, dove $\NN[x]$ denota l'insieme dei polinomi a coefficienti in $\NN$, e definiamo:
	\[ p \prec q \;\;\Mydef\;\; \exists {N \in \NN} \; \forall {x \in
\NN} \; x > N \rightarrow p(x) < q(x)
		\]
	ossia $p \prec q$ se $p(x)<q(x)$ definitivamente.
\end{exercise}

\begin{proposition}
	Se $(A,<_A)$ e $(B,<_B)$ sono buoni ordini, allora anche:
	\[ (A,<_A) + (B,<_B) \qquad (A,<_A) \cdot (B,<_B) \qquad (A,<_A)^{(B,<_B)}
		\]
	sono buoni ordini.
\end{proposition}

\begin{proof}
	Si tratta di facili verifiche, {\drkred eccetto la terza.}

	La relazione $<_{\exp}$ \`e {\drkblu irriflessiva} per definizione.
Verifichiamo la {\drkblu transitivit\`a.}
	Assumiamo $f <_{\exp} g$ e $g <_{\exp} h$ con
	\begin{align*}
		m_{fg} &= \max \{x\in B\: |\: f(x) \ne g(x)\} \\
		m_{gh} &= \max \{x\in B\: |\: g(x) \ne h(x)\} \\
		m_{fh} &= \max \{x\in B\: |\: f(x) \ne h(x)\}
	\end{align*}
	detto $m = \max(m_{fg},m_{gh})$, abbiamo $f(m) \leq g(m) \leq
h(m)$, dove la prima disuguaglianza \`e stretta se $m = m_1$, e la seconda
lo \`e se $m = m_2$, in ogni caso, almeno una delle due disuguaglianze \`e
stretta, per cui abbiamo $f(m) < h(m)$.
	D'altro canto, se $m < x$, allora $f(x) = g(x) = h(x)$, per cui $m
= m_{fh}$, e, di conseguenza, $<_{\exp}$ \`e transitiva.

La relazione $<_{\exp}$ \`e un {\drkblu ordine totale} perch\'e, se $f \ne
g$, allora \`e ben definito
\[m = \max\{x \in B \:|\: f(x) \ne g(x)\}\]
per cui o $f(m)< g(m)$ o $g(m) < f(m)$: nel primo caso $f <_{\exp} g$, nel secondo $g <_{\exp} f$.

	Resta da verificare che {\drkblu l'ordine $<_{\exp}$ \`e buono.}
Procedendo per assurdo, supponiamo che l'insieme
$S$ delle funzioni a supporto finito da $B$ ad $A$ non sia bene ordinato
da~$<_{\exp}$. Allora
	\[ \underbrace{\exists {f \in S} \; \exists {A \subseteq S} (f \in
A}_{\text{c'\`e un $A \subseteq S$ non vuoto}}\land\underbrace{\forall {g
\in A} \; \exists {h \in A} \; h <_{\exp} g}_{\text{che non ha minimo}})
		\]
Ora fissiamo una particolare $f \in S$ tale che $\exists A \subseteq S
\text{\etc.}$ con queste propriet\`a:
	che $m=\max(\supp(f))$ sia il minimo, e che, a parit\`a di $\max(\supp(f))$, il valore di $f(m)$ sia minimo.
	Sia $A$ tale che $f \in A$ ed $A$ \`e un sottoinsieme senza minimo.
	Il nostro scopo \`e costruire $\widetilde{A} \subseteq S$ che non
abbia minimo e contenga una funzione $\widetilde{f}$ con $\widetilde{m} =
\max(\supp(f)) < m$, contro la minimalit\`a di $m$.

	Osserviamo innanzitutto che $A$ pu\`o essere ripartito in:
	\begin{align*}
		A_1 &= \{g \in A |\max(\supp(g)) < m\} \\
		A_2 &= \{g \in A |\max(\supp(g)) = m \land g(m) = f(m)\} \\
		A_3 &= \{g \in A |\max(\supp(g)) = m \land f(m) < g(m)\} \\
		A_4 &= \{g \in A |m < \max(\supp(g))\}
	\end{align*}
	Le funzioni in $A_1$ sono $<_{\exp}$ di quelle in $A_2$, che sono
$<_{\exp}$ \etc. fino ad $A_4$
	Per\`o $A_1$ \`e vuoto, perch\'e altrimenti, presa $f' \in A_1$, abbiamo $f' \in A$ e $\max(\supp(f))<m$ contro la minimalit\`a di $m$.
	Abbiamo invece che $A_2$ non \`e vuoto perch\'e contiene $f$, e
non ha minimo, infatti, se avesse minimo, questo dovrebbe essere anche il
minimo di $A$.

	Concentriamoci su $A_2$. Tutte le $g \in A_2$ assumono il medesimo valore su $m$, quindi, comparando due di queste funzioni con $<_{\exp}$, il valore assunto da entrambe su $m$ 
	\`e irrilevante. Per cui la funzione:
	\begin{align*} H \colon A_2 &\to S \\
g &\mapsto \widehat{g} \quad\text{con}\quad\widehat{g}(x) = \smash{\begin{cases}
		\min A &\text{se $x = m$}\\
		g(x) &\text{altrimenti}
	\end{cases}}
		\end{align*}
\vskip.5\baselineskip
	\`e strettamente crescente.
Per cui $\widetilde{A} = H[A_2]$ non ha minimo.
	Ora, per\`o, fissata $g \in A_2$, abbiamo $\supp(\widehat{g}) =
\supp(g) \setminus \{m\}$, quindi, ponendo $\widetilde{f} = \widehat{g}$,
abbiamo $\max(\supp(\widetilde{f}))<m$, e questo contraddice la
minimalit\`a di~$m$.
\end{proof}

\begin{proposition}\label{op-bo-bd}
	Le operazioni aritmetiche sui buoni ordini \textit{passano al
quoziente modulo isomorfismi}. Ossia, dati due buoni ordini $\mathcal A = (B,<_A)$
	e $\mathcal B = (B,<_B)$, e dati $\mathcal A' = (A',<_{A'}) \sim \mathcal A$ e $\mathcal{B}' = (B',<_{B'}) \sim \mathcal B$, si ha:
	\[ \mathcal{A} + \mathcal{B} \;\sim\; \mathcal{A}' + \mathcal{B'}
\qquad \mathcal{A} \cdot \mathcal{B} \;\sim\; \mathcal{A}' \cdot
\mathcal{B'} \qquad \mathcal{A}^{\mathcal{B}} \;\sim\; \mathcal{A}'^{\mathcal{B'}}
		\]
\end{proposition}

\begin{proof}
	Fissati gli isomorfismi $f\colon A \to A'$ e $g \colon B \to B'$, \`e facile scrivere esplicitamente gli isomorfismi richiesti. Per esempio, nel caso di $\mathcal{A}^{\mathcal{B}}$, si considera la restrizione alle funzioni a supporto finito di:
	\begin{align*}
{}^{B}A &\to {}^{B'}A'\\
h &\mapsto f \circ h \circ g^{-1}
		\end{align*}
\end{proof}

\begin{exercise}
Svolgere qualcuna delle verifiche nella dimostrazione della proposizione~\ref{op-bo-bd}.
\end{exercise}

\begin{proposition}[Propriet\`a delle operazioni sui buoni
ordini]\label{op-bo-pro}
	Siano $\mathcal{A} = (A,<_A)$, $\mathcal{B} = (B,<_B)$ e $\mathcal{C} = (C,<_C)$ buoni ordini. Allora:
\begin{alignat*}{3}
(\mathcal A + \mathcal B) + \mathcal C \;&\sim\; \mathcal A +
(\mathcal B + \mathcal C) \quad &&\quad &(\mathcal A \cdot \mathcal B) \cdot
\mathcal C \;&\sim\; \mathcal A \cdot (\mathcal B \cdot \mathcal C)\\
&&&\mathclap{\mathcal A \cdot (\mathcal B + \mathcal C) \;\sim\;
\mathcal A \cdot \mathcal B + \mathcal A \cdot \mathcal C} \\
{\mathcal A}^{\mathcal B + \mathcal C} \;&\sim\; \mathcal A^{\mathcal
B} \cdot {\mathcal A}^{\mathcal C} &&& ({\mathcal A}^{\mathcal
B})^{\mathcal{C}} \;&\sim\; {\mathcal A}^{\mathcal B \cdot \mathcal C}
\end{alignat*}
\end{proposition}

\begin{proof}
	Facili verifiche.
\end{proof}

\begin{exercise}
Svolgere qualcuna delle verifiche nella dimostrazione della
proposizione~\ref{op-bo-pro}.
\end{exercise}

\`E importante notare che non tutte le propriet\`a delle operazioni aritmetiche su $\omega$ valgono per i buoni ordini.

\begin{exercise}
	Esibisci {\drkred controesempi} alle seguenti:
	{\drkred \begin{align*}
		\mathcal A + \mathcal B \;&\sim\; \mathcal B + \mathcal A
&(\mathcal A + \mathcal B) \cdot \mathcal C \;&\sim\; \mathcal A \cdot \mathcal C + \mathcal B \cdot \mathcal C \\
		\mathcal A \cdot \mathcal B \;&\sim\; \mathcal B \cdot
\mathcal A &(\mathcal A \cdot \mathcal B)^{\mathcal C} \;&\sim\; \mathcal A^{\mathcal C} \cdot \mathcal B^{\mathcal C}
	\end{align*}}

\vskip-3\baselineskip
\ 
\end{exercise}
Un altro tranello in cui si potrebbe cadere \`e credere che le operazioni
sui buoni ordini estendano quelle sulle cardinalit\`a. Questo \`e vero per le cardinalit\`a finite, dove le operazioni di $\omega$ coincidono con quelle cardinali e ordinali, e anche in generale 
per somma e prodotto -- come \`e chiaro dalla definizione -- ma fallisce
per l'elevamento a potenze infinite.

\begin{exercise}
	Dimostra che se $\mathcal A = (A,<_A)$ e $\mathcal B = (B,<_B)$ sono buoni ordini con $|A| = |B| = \aleph_0$ e $(C,<_C) = \mathcal A^{\mathcal B}$ allora $|C| = \aleph_0$.
\end{exercise}
