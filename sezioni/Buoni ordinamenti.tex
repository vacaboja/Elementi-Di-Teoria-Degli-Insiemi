\section{I buoni ordini}
Il nostro prossimo obiettivo \`e definire e studiare la classe dei \vocab{numeri ordinali}. Questa pu\`o essere pensata come la pi\`u vasta classe
-- dotata di un ordine totale definito per mezzo di una formula -- su cui sia corretto ragionare per induzione forte. Conteremo, quindi, sugli ordinali per formulare 
l'induzione e la ricorsione transfinita, procedimenti che superano la forza dimostrativa dell'induzione e della ricorsione aritmetica -- per esempio permettendo di ottenere 
il teorema di Cantor-Lebesgue sugli insiemi di unicit\`a.

Siccome l'induzione forte equivale al principio del minimo, studieremo i buoni ordini. In questa sezione, dimostreremo il risultato seguente.

\begin{theorem}
	Siano $(A,<_A)$ e $(B,<_B)$ {\drkgrn -- scriviamo semplicemente $(A,<)$
e~$(B,<)$, per comodit\`a --} insiemi bene ordinati. Allora vale una e una
sola delle seguenti:
	\begin{itemize}
		\item $(A,<)$ \`e isomorfo a un segmento iniziale proprio di $(B,<)$
		\item $(A,<)$ e $(B,<)$ sono isomorfi
		\item $(B,<)$ \`e isomorfo a un segmento iniziale di $(A,<)$
	\end{itemize}

Quindi, definendo
\[ (A,<) \prec (B,<) \;\;\Mydef\;\; \exists C \; \text{$C$ \`e segmento
iniziale proprio di $(B,<)$ e $(A,<) \sim C$}
	\]
$\prec$ soddisfa le propriet\`a formali di un ordine totale fra le classi di isomorfismo di buoni ordini.
\end{theorem}

\begin{remark}
L'ordine largo associato a $\prec$ \`e:
\[ (A,<) \preceq (B,<) \;\;\Mydef\;\; (A,<) \prec (B,<) \lor (A,<) \sim (B,<)
	\]
ossia ``$(A,<)$ \`e isomorfo a un segmento iniziale di $(B,<)$''.
\end{remark}

Richiamiamo alcune definizioni fondamentali.

\begin{definition}[Buon ordine]
	$(A,<)$ \`e un \vocab{buon ordine} se ogni $B \subseteq A$ non vuoto ha un minimo elemento.
\end{definition}

\begin{definition}[Segmento inziale]
	Dato un ordine totale $(A,<)$, $B \subseteq A$ \`e un
\vocab{segmento inziale} di~$A$ se
\[\forall {b \in B} \; \forall {x \in A} \; x < b \rightarrow x \in B\]
\end{definition}

\begin{definition}[Segmenti iniziali propri e principali]
	Il segmento iniziale $B$ \`e \vocab{proprio} se $B \ne A$. Il segmento iniziale $B$ \`e \vocab{principale} se \`e della forma:
	\[ B \;=\; A_a \;\Mydef\; \{x \in A | x < a\}
		\]
	per qualche $a \in A$, e, in questo caso, si dice  che \`e il \vocab{segmento iniziale principale determinato da $a$}.
\end{definition}

Un segmento iniziale principale, $A_a$, \`e automaticamente proprio,
perch\'e $a \not \in A_a$. Nel caso dei buoni ordini questa \`e una doppia
implicazione:

\begin{proposition}
	Ogni segmento iniziale proprio di un buon ordine \`e principale.
\end{proposition}

\begin{proof}
	Sia $(A,<)$ ben ordinato e $I \subsetneq A$ un segmento iniziale
proprio. Consideriamo $a = \min (A \setminus I)$. Allora $I = A_a$.

	{\drkgrn Verifiche. $x\in A_a \Rightarrow x < a \Rightarrow
x\not\in A\setminus I \Rightarrow x\in I$.\\
Viceversa, se $x\in I$, non pu\`o valere $a\le x$, perch\'e $a\not\in I$,
quindi $x<a$, cio\`e $x\in A_a$.}
\end{proof}

\begin{exercise}
	Dimostra che la proposizione precedente caratterizza i buoni ordini. Pi\`u precisamente, dato un ordine totale $(A,<)$, se ogni segmento iniziale proprio di $A$
	\`e principale, allora $A$ \`e bene ordinato da $<$.
\end{exercise}

\begin{exercise}
Perch\'e non \`e vero che, in~$\ZZ$, ogni segmento iniziale proprio \`e
principale?
\end{exercise}

\begin{lemma}
	Sia $(A,<)$ un buon ordine e $f \colon A \to A$ una funzione
strettamente crescente, allora $\forall {x \in A} \; x \leq f(x)$.
\end{lemma}

\begin{proof}
	Per assurdo, assumiamo la negazione della tesi. Quindi l'insieme $B = \{x \in A | f(x) < x\}$ non \`e vuoto. Sia $k = \min B$.
	Allora $f(k) < k$ per la definizione di~$B$. Applicando $f$ ad
ambo i membri, siccome $f$ \`e
crescente, $f(f(k))<f(k)$, ossia $f(k) \in B$. Ricordando che $f(k)<k$, per\`o,
questo contraddice la minimalit\`a di~$k$.
\end{proof}

\begin{corollary}\label{cor-bonanza}
\ 
\vskip-2.5\baselineskip
\ 
	\begin{enumerate}[(1)]
		\item Un buon ordine non \`e isomorfo a un suo segmento iniziale proprio.
		\item L'identit\`a \`e il solo isomorfismo fra un buon ordine e se stesso.
		\item Se $(A,<)$ e $(B,<)$ sono buoni ordini isomorfi allora esiste un unico isomorfismo fra di essi.
	\end{enumerate}
\end{corollary}

\begin{proof}\ 

\vskip-2\baselineskip
	\begin{enumerate}[(1)]
		\item Sia $f \colon A \to A_a$ un isomorfismo.
		In particolare $f$ \`e crescente. Tuttavia $f(a) <
a\;\lightning$
		\item Sia $f \colon A \to A$ un automorfismo del buon
ordine $(A,<)$. Allora sia $f$ sia $f^{-1}$ sono crescenti. Quindi, per
ogni $x\in A$, valgono
$x\le f(x)$ e $x\le f^{-1}(x)$. Dalla seconda, per\`o, segue $f(x)\le x$,
che, combinata con la prima, implica $f(x)=x$.
		\item Se $f \colon A \to B$ e $g \colon A \to B$ fossero
due isomorfismi diversi, allora $g^{-1} \circ f \colon A \to A$ sarebbe un
automorfismo di $A$ diverso dall'identit\`a.
	\end{enumerate}
\end{proof}

\begin{remark}
	Siano $(A,<)$, $(B,<)$, $(C,<)$ buoni ordini, allora:
	\[ (A,<) \preceq (B,<) \;\land\; (B,<) \preceq (C,<)
\;\;\rightarrow\;\; (A,<) \preceq (C,<)
		\]
\end{remark}

\begin{proof}
	Siano $f\colon A \to B$ e $g \colon B \to C$ isomorfismi fra $A$ e un segmento iniziale di $B$ e fra $B$ e un segmento iniziale di $C$ rispettivamente.
	Dimostriamo che $g \circ f \colon A \to C$ \`e un isomorfismo fra $A$ e un segmento iniziale di $C$. Naturalmente $g \circ f$ \`e crescente in quanto composizione di funzioni crescenti, dunque occorre solo verificare che $g \circ f[A]$ \`e un segmento iniziale di $C$.

{\drkgrn Verifica. Sia $g(f(a))$ un qualunque elemento di $g \circ f[A]$, e sia $x \in C$ tale che $x < g(f(a))$, dobbiamo verificare che $x \in g \circ f[A]$.
	Siccome $g(f(a)) \in g[B]$ e, per ipotesi, $g[B]$ \`e un segmento
iniziale di $C$, otteniamo $x \in g[B]$. Possiamo allora scrivere $x =
g(y)$. Ora, siccome $g$ \`e un isomorfismo, da $x = g(y) < g(f(a))$
deduciamo $y < f(a)$, inoltre, per ipotesi, $f[A]$
	\`e un segmento iniziale di $B$, pertanto $y \in f[A]$, ovvero $y = f(z)$ per $z \in A$. Abbiamo quindi $x = g(f(z)) \in g \circ f[A]$.}
\end{proof}

\begin{remark}
	Siano $(A,<)$ e $(B,<)$ buoni ordini, allora:
	\[ (A,<) \preceq (B,<) \;\land\; (B,<) \preceq (A,<)
\;\;\rightarrow\;\; (A,<) \sim (B,<)
		\]
\end{remark}

\begin{proof}
	Siano $f \colon A \to B$ e $g \colon B \to A$ isomorfismi fra $A$ e un segmento iniziale di $B$ e fra $B$ e un segmento iniziale di $A$.
	Ricordando la dimostrazione dell'osservazione precedente, $g \circ
f$ \`e un isomorfismo fra $A$ e un segmento iniziale $g \circ f[A]$ di
$A$, ma per il corollario~\ref{cor-bonanza}~(1)
	$g \circ f[A]$ non pu\`o essere un segmento iniziale proprio di
$A$, quindi $g\circ f[A]=A$.
	Per il corollario~\ref{cor-bonanza}~(2), $g \circ f = \id_A$.
Ragionando simmetricamente si ottiene $f \circ g = \id_B$.
\end{proof}

Possiamo finalmente passare alla dimostrazione del teorema.

\begin{theorem}\label{bo-trico-teo}
	Siano $(A,<)$ e $(B,<)$ insiemi ben ordinati, allora vale una e una sola delle seguenti:
	\[ (A,<) \prec (B,<) \qquad (A,<) \sim (B,<) \qquad (B,<) \prec (A,<)
		\]
\end{theorem}

\begin{wrapfigure}[16]{l}{.3\textwidth}
	\includegraphics[width = .3\textwidth]{immagini/ordine_totale_buoni_ordini.png}
\end{wrapfigure}

{\drkgrn Idea: Il corollario ci dice che vale al pi\`u una delle alternative, quindi la difficolt\`a risiede nel dimostrare che una si verifica. Molto vagamente potremmo ragionare cos\`i.
Identifichiamo, progressivamente, segmenti iniziali sempre pi\`u lunghi di
$A$ e $B$. All'inizio identifichiamo il minimo di $A$ con il minimo di
$B$, poi il secondo elemento di $A$ con il secondo elemento di $B$, \etc. Fatti $\omega$
passi avremo identificato un segmento iniziale di $A$, diciamo $A_x$, isomorfo a $\omega$, con un $B_y$, anch'esso ovviamente isomorfo a $\omega$. Bene: continuiamo identificando $x$ con $y$. Quando potrebbe bloccarsi il procedimento?
Solo se, ad un certo punto, abbiamo identificato interamente uno dei due
insiemi, con un segmento iniziale dell'altro -- perch\'e altrimenti, abbiamo identificato due segmenti iniziali $A_x$ e $B_y$ e possiamo continuare attaccando $x$ a $y$. \`E 
come la chiusura di una cerniera lampo: ad ogni istante c'\`e un prossimo dente.

{\drkred Questo discorso, per\`o, non \`e una dimostrazione.} Se vogliamo,
sarebbe un tentativo di costruire l'isomorfismo cercato per
\vocab{ricorsione transfinita.} Il guaio \`e che i numeri che permetterebbero di numerare i passaggi della costruzione,
gli \vocab{ordinali,} sono appunto l'oggetto che stiamo tentando di costruire.}

\begin{proof}
	Per il corollario visto prima, si pu\`o verificare al pi\`u una
delle tre condizioni. Dimostriamo dunque che se ne verifica almeno una.
Consideriamo l'insieme $f$ cos\`i definito
	\[ f \;\;=\;\; \{(a,b) \in A \times B | A_a \sim B_b\}
		\]
	Vogliamo dimostrare che $f$ \`e una funzione, che \`e crescente,
che $\Dom(f)$ \`e un segmento iniziale di $A$, e che $\Imm(f)$ \`e un
segmento iniziale di $B$. Quindi $f$ \`e un isomorfismo fra un segmento iniziale di $A$ e uno di $B$.
	Infine dimostreremo che $\Dom(f) = A$ o $\Imm(f) = B$, e questo
conclude la dimostrazione.

{\drkblu $f$ \`e una funzione.} Supponiamo per assurdo $(a,b) \in f$ e
$(a,b') \in f$ con $b \ne b'$. Per la costruzione di~$f$ abbiamo:
		\[ B_b \sim A_a \sim B_{b'}
			\]
D'altro canto, senza perdita di generalit\`a, possiamo supporre $b < b'$, allora $B_b$ \`e un
segmento iniziale proprio di~$B_{b'}$, contraddicendo il fatto
che~$B_b\sim B_{b'}$.

{\drkblu $f$ \`e crescente.} Dati $a,a' \in A$, con $a<a'$, dobbiamo dimostrare $f(a) < f(a')$. Supponiamo, per assurdo $f(a') \leq f(a)$, abbiamo allora:
		\[ A_{a'} \sim B_{f(a')} \preceq B_{f(a)} \sim A_a \prec
A_{a'}
			\]
dove i due isomorfismi seguono dalla costruzione di~$f$, mentre le due
disuguaglianze seguono dalle inclusioni $B_{f(a')} \subseteq B_{f(a)}$ e
$\sim A_a\subsetneq  A_{a'}$.
Abbiamo ottenuto $A_{a'}\prec A_{a'}\;\lightning$

{\drkblu $\Dom(f)$ \`e s.i.\ di $A$.} Sia $a \in \Dom(f)$ e $a' < a$,
vogliamo dimostrare che $a' \in \Dom(f)$. Osserviamo che $A_{a'}$ \`e un
segmento iniziale proprio di $A_a$. L'isomorfismo che attesta $A_a\sim
B_{f(a)}$, manda quindi $A_{a'}$ in un segmento iniziale proprio di
$B_{f(a)}$. Questo \`e, in particolare, un segmento iniziale proprio
$B_{b'}$ di~$B$. Quindi $(a',b')\in f$.

{\drkblu $\Imm(f)$ \`e s.i. di $B$.} Dimostrazione simmetrica alla precedente.

{\drkblu $\Dom(f) = A$ o $\Imm(f) = B$.} Se cos\`i non fosse, avremmo $\Dom(f) = A_a$ e $\Imm(f) = B_b$
		per opportuni $a \in A$ e $b \in B$. Siccome $f$ \`e
crescente $A_a = \Dom(f) \sim \Imm(f) = B_b$.  Abbiamo quindi $A_a \sim
B_b$, per cui $(a,b)\in f\;\lightning$
\end{proof}

\begin{exercise}
	Sia $(A,<)$ un buon ordine e sia $B \subsetneq A$. Dimostra che $B \preceq A$, ma non necessariamente $B \prec A$.
\end{exercise}

\begin{exercise}
	Sia $(A,<)$ un ordine totale con $A = \union S$. Supponiamo che:
	\begin{enumerate}[1.]
		\item ogni $X \in S$ \`e bene ordinato dalla restrizione
di~$<$ a~$X$
		\item per ogni $X,Y \in S$, o $X$ \`e segmento iniziale di $Y$ o $Y$ \`e segmento iniziale di $X$
	\end{enumerate}
	Dimostra che $(A,<)$ \`e un buon ordine. Esibisci inoltre un controesempio eliminando la condizione 2 dalle ipotesi.
\end{exercise}

\subsection{Operazioni aritmetiche fra buoni ordini}
Per ora, non abbiamo visto molti esempi di buoni ordini. Le operazione definite in questa sezione forniscono una prima sorgente di esempi concreti.
Nel seguito del corso, vedremo buoni ordini assai pi\`u vasti di quelli ottenibili con queste operazioni.

\begin{definition}[Somma di ordini totali]
	Dati $(A,<_A)$ e $(B,<_B)$ ordini totali. Definiamo
	\[ {\drkblu (A,<_A) + (B,<_B)} \;\;\Mydef\;\; (A \sqcup B, <_+)
		\]
	dove, ricordiamo che $A \sqcup B = (A \times\{0\}) \cup (B \times \{1\})$, e $<_+$ \`e definito da:
	\begin{align*}
		{\drkblu (x,y) <_+ (x',y')} \;\;\Mydef&\phantom{\lor}\; (y = 0 \land y' = 1) \\
						  &{\lor}\; (y = 0 \land y' = 0 \land x <_A x') \\
						  &{\lor}\; (y = 1 \land y' = 1 \land x <_B x')
	\end{align*}
\end{definition}

{\drkgrn Idea: $(A,<_A) + (B,<_B)$ si ottiene attaccando $(B,<_B)$ in coda a $(A,<_A)$}.

\begin{figure}[H]
	\centering
	\includegraphics[width = .7\textwidth]{immagini/somma_ordini_totali.png}
\end{figure}

Ripetiamo, per completezza, la definizione di prodotto lessicografico.

\begin{definition}[Prodotto di lessicografico]
	Siano $(A,<_A)$ e $(B,<_B)$ ordini totali. Definiamo
	\[ {\drkblu (A,<_A) \cdot (B,<_B)} \;\;\Mydef\;\; (A \times B, <_\times)
		\]
	dove $<_\times$ \`e definito da:
	\[ {\drkblu (x,y) <_\times (x',y')} \;\;\Mydef\;\; (y <_B y') \lor (y = y' \land x <_A x')
		\]
\end{definition}

{\drkgrn Idea: confronto prima la seconda componente; a parit\`a conta la
prima. Quindi $(A,<_A) \cdot (B,<_B)$ pu\`o essere pensato come tante
copie di~$(A,<_A)$ messe, fra loro, nello stesso ordine degli elementi
di~$(B,<_B)$: in breve, $(A,<_A)$ ripetuto $(B,<_B)$~volte.}

\begin{figure}[H]
	\centering
	\includegraphics[width = \textwidth]{immagini/ordine_lessicografico.png}
\end{figure}

Per definire l'esponenziale ci serve la nozione di supporto.

\begin{definition}
	Dato un buon ordine $(B,<)$ e $f\colon A \to B$, il \vocab{supporto} di $f$ \`e:
	\[ {\drkblu\supp_B(f)} \;\;\Mydef\;\; \{\:x \in A\: |\: f(x) \ne \min_B\:\}
		\]
	Scriveremo semplicemente $\drkblu\supp(f)$ quando \`e chiaro cosa sia $B$.
\end{definition}

\begin{definition}[Potenza di ordini totali]
	Dati $(A,<_A)$ e $(B,<_B)$ ordini totali, definiamo
	\[ {\drkblu (A,<_A)^{(B,<_B)}} \;\;\Mydef\;\; \Big(\;\big\oldparen
\underbrace{f \in {}^{B}A \: \big| \: |\supp(f)\,| <
\aleph_0}_{\text{funzioni $B\to A$ a supporto finito}}\big\}\;,\;<_{\exp}\;\Big)
		\]
	dove $<_{\exp}$ \`e definito da:
	\begin{align*} {\drkblu f <_{\exp} g} \;\;\Mydef\;\; f \ne
g\;\;\text{e}\;\;&\text{detto $m=\max\{x\in B|f(x)\neq g(x)\}$}\\
&\text{vale $f(m) < g(m)$}
\end{align*}
\end{definition}

{\drkgrn L'idea \`e che una funzione $B \rightarrow A$ pu\`o essere vista
come una specie di tupla con tante componenti quanti sono gli elementi di
$B$.
Ordinare due tuple lessicograficamente significa confrontare
la componente corrispondente all'elemento di $B$ pi\`u
grande su cui le funzioni siano diverse.}

\begin{exercise}
	Verifica che $(\omega,<)^{(\omega,<)} \sim (\NN[x],\prec)$, dove $\NN[x]$ denota l'insieme dei polinomi a coefficienti in $\NN$, e definiamo:
	\[ p \prec q \;\;\Mydef\;\; \exists {N \in \NN} \; \forall {x \in
\NN} \; x > N \rightarrow p(x) < q(x)
		\]
	ossia $p \prec q$ se $p(x)<q(x)$ definitivamente.
\end{exercise}

\begin{proposition}
	Se $(A,<_A)$ e $(B,<_B)$ sono buoni ordini, allora anche:
	\[ (A,<_A) + (B,<_B) \qquad (A,<_A) \cdot (B,<_B) \qquad (A,<_A)^{(B,<_B)}
		\]
	sono buoni ordini.
\end{proposition}

\begin{proof}
	Si tratta di facili verifiche, {\drkred eccetto la terza.}

	La relazione $<_{\exp}$ \`e {\drkblu irriflessiva} per definizione.
Verifichiamo la {\drkblu transitivit\`a.}
	Assumiamo $f <_{\exp} g$ e $g <_{\exp} h$ con
	\begin{align*}
		m_{fg} &= \max \{x\in B\: |\: f(x) \ne g(x)\} \\
		m_{gh} &= \max \{x\in B\: |\: g(x) \ne h(x)\} \\
		m_{fh} &= \max \{x\in B\: |\: f(x) \ne h(x)\}
	\end{align*}
	detto $m = \max(m_{fg},m_{gh})$, abbiamo $f(m) \leq g(m) \leq
h(m)$, dove la prima disuguaglianza \`e stretta se $m = m_1$, e la seconda
lo \`e se $m = m_2$, in ogni caso, almeno una delle due disuguaglianze \`e
stretta, per cui abbiamo $f(m) < h(m)$.
	D'altro canto, se $m < x$, allora $f(x) = g(x) = h(x)$, per cui $m
= m_{fh}$, e, di conseguenza, $<_{\exp}$ \`e transitiva.

La relazione $<_{\exp}$ \`e un {\drkblu ordine totale} perch\'e, se $f \ne
g$, allora \`e ben definito
\[m = \max\{x \in B \:|\: f(x) \ne g(x)\}\]
per cui o $f(m)< g(m)$ o $g(m) < f(m)$: nel primo caso $f <_{\exp} g$, nel secondo $g <_{\exp} f$.

	Resta da verificare che {\drkblu l'ordine $<_{\exp}$ \`e buono.}
Procedendo per assurdo, supponiamo che l'insieme
$S$ delle funzioni a supporto finito da $B$ ad $A$ non sia bene ordinato
da~$<_{\exp}$. Allora
	\[ \underbrace{\exists {f \in S} \; \exists {A \subseteq S} (f \in
A}_{\text{c'\`e un $A \subseteq S$ non vuoto}}\land\underbrace{\forall {g
\in A} \; \exists {h \in A} \; h <_{\exp} g}_{\text{che non ha minimo}})
		\]
Ora fissiamo una particolare $f \in S$ tale che $\exists A \subseteq S
\text{\etc.}$ con queste propriet\`a:
	che $m=\max(\supp(f))$ sia il minimo, e che, a parit\`a di $\max(\supp(f))$, il valore di $f(m)$ sia minimo.
	Sia $A$ tale che $f \in A$ ed $A$ \`e un sottoinsieme senza minimo.
	Il nostro scopo \`e costruire $\widetilde{A} \subseteq S$ che non
abbia minimo e contenga una funzione $\widetilde{f}$ con $\widetilde{m} =
\max(\supp(f)) < m$, contro la minimalit\`a di $m$.

	Osserviamo innanzitutto che $A$ pu\`o essere ripartito in:
	\begin{align*}
		A_1 &= \{g \in A |\max(\supp(g)) < m\} \\
		A_2 &= \{g \in A |\max(\supp(g)) = m \land g(m) = f(m)\} \\
		A_3 &= \{g \in A |\max(\supp(g)) = m \land f(m) < g(m)\} \\
		A_4 &= \{g \in A |m < \max(\supp(g))\}
	\end{align*}
	Le funzioni in $A_1$ sono $<_{\exp}$ di quelle in $A_2$, che sono
$<_{\exp}$ \etc. fino ad $A_4$
	Per\`o $A_1$ \`e vuoto, perch\'e altrimenti, presa $f' \in A_1$, abbiamo $f' \in A$ e $\max(\supp(f))<m$ contro la minimalit\`a di $m$.
	Abbiamo invece che $A_2$ non \`e vuoto perch\'e contiene $f$, e
non ha minimo, infatti, se avesse minimo, questo dovrebbe essere anche il
minimo di $A$.

	Concentriamoci su $A_2$. Tutte le $g \in A_2$ assumono il medesimo valore su $m$, quindi, comparando due di queste funzioni con $<_{\exp}$, il valore assunto da entrambe su $m$ 
	\`e irrilevante. Per cui la funzione:
	\begin{align*} H \colon A_2 &\to S \\
g &\mapsto \widehat{g} \quad\text{con}\quad\widehat{g}(x) = \smash{\begin{cases}
		\min A &\text{se $x = m$}\\
		g(x) &\text{altrimenti}
	\end{cases}}
		\end{align*}
\vskip.5\baselineskip
	\`e strettamente crescente.
Per cui $\widetilde{A} = H[A_2]$ non ha minimo.
	Ora, per\`o, fissata $g \in A_2$, abbiamo $\supp(\widehat{g}) =
\supp(g) \setminus \{m\}$, quindi, ponendo $\widetilde{f} = \widehat{g}$,
abbiamo $\max(\supp(\widetilde{f}))<m$, e questo contraddice la
minimalit\`a di~$m$.
\end{proof}

\begin{proposition}\label{op-bo-bd}
	Le operazioni aritmetiche sui buoni ordini \textit{passano al
quoziente modulo isomorfismi}. Ossia, dati due buoni ordini $\mathcal A = (B,<_A)$
	e $\mathcal B = (B,<_B)$, e dati $\mathcal A' = (A',<_{A'}) \sim \mathcal A$ e $\mathcal{B}' = (B',<_{B'}) \sim \mathcal B$, si ha:
	\[ \mathcal{A} + \mathcal{B} \;\sim\; \mathcal{A}' + \mathcal{B'}
\qquad \mathcal{A} \cdot \mathcal{B} \;\sim\; \mathcal{A}' \cdot
\mathcal{B'} \qquad \mathcal{A}^{\mathcal{B}} \;\sim\; \mathcal{A}'^{\mathcal{B'}}
		\]
\end{proposition}

\begin{proof}
	Fissati gli isomorfismi $f\colon A \to A'$ e $g \colon B \to B'$, \`e facile scrivere esplicitamente gli isomorfismi richiesti. Per esempio, nel caso di $\mathcal{A}^{\mathcal{B}}$, si considera la restrizione alle funzioni a supporto finito di:
	\begin{align*}
{}^{B}A &\to {}^{B'}A'\\
h &\mapsto f \circ h \circ g^{-1}
		\end{align*}
\end{proof}

\begin{exercise}
Svolgere qualcuna delle verifiche nella dimostrazione della proposizione~\ref{op-bo-bd}.
\end{exercise}

\begin{proposition}[Propriet\`a delle operazioni sui buoni
ordini]\label{op-bo-pro}
	Siano $\mathcal{A} = (A,<_A)$, $\mathcal{B} = (B,<_B)$ e $\mathcal{C} = (C,<_C)$ buoni ordini. Allora:
\begin{alignat*}{3}
(\mathcal A + \mathcal B) + \mathcal C \;&\sim\; \mathcal A +
(\mathcal B + \mathcal C) \quad &&\quad &(\mathcal A \cdot \mathcal B) \cdot
\mathcal C \;&\sim\; \mathcal A \cdot (\mathcal B \cdot \mathcal C)\\
&&&\mathclap{\mathcal A \cdot (\mathcal B + \mathcal C) \;\sim\;
\mathcal A \cdot \mathcal B + \mathcal A \cdot \mathcal C} \\
{\mathcal A}^{\mathcal B + \mathcal C} \;&\sim\; \mathcal A^{\mathcal
B} \cdot {\mathcal A}^{\mathcal C} &&& ({\mathcal A}^{\mathcal
B})^{\mathcal{C}} \;&\sim\; {\mathcal A}^{\mathcal B \cdot \mathcal C}
\end{alignat*}
\end{proposition}

\begin{proof}
	Facili verifiche.
\end{proof}

\begin{exercise}
Svolgere qualcuna delle verifiche nella dimostrazione della
proposizione~\ref{op-bo-pro}.
\end{exercise}

\`E importante notare che non tutte le propriet\`a delle operazioni aritmetiche su $\omega$ valgono per i buoni ordini.

\begin{exercise}
	Esibisci {\drkred controesempi} alle seguenti:
	{\drkred \begin{align*}
		\mathcal A + \mathcal B \;&\sim\; \mathcal B + \mathcal A
&(\mathcal A + \mathcal B) \cdot \mathcal C \;&\sim\; \mathcal A \cdot \mathcal C + \mathcal B \cdot \mathcal C \\
		\mathcal A \cdot \mathcal B \;&\sim\; \mathcal B \cdot
\mathcal A &(\mathcal A \cdot \mathcal B)^{\mathcal C} \;&\sim\; \mathcal A^{\mathcal C} \cdot \mathcal B^{\mathcal C}
	\end{align*}}

\vskip-3\baselineskip
\ 
\end{exercise}
Un altro tranello in cui si potrebbe cadere \`e credere che le operazioni
sui buoni ordini estendano quelle sulle cardinalit\`a. Questo \`e vero per le cardinalit\`a finite, dove le operazioni di $\omega$ coincidono con quelle cardinali e ordinali, e anche in generale 
per somma e prodotto -- come \`e chiaro dalla definizione -- ma fallisce
per l'elevamento a potenze infinite.

\begin{exercise}
	Dimostra che se $\mathcal A = (A,<_A)$ e $\mathcal B = (B,<_B)$ sono buoni ordini con $|A| = |B| = \aleph_0$ e $(C,<_C) = \mathcal A^{\mathcal B}$ allora $|C| = \aleph_0$.
\end{exercise}

\subsection{Gli ordinali di Von Neumann}
In questa sezione definiremo gli ordinali di Von Neumann. L'idea che vogliamo concretizzare \`e che, siccome abbiamo visto che, 
a meno di isomorfismi, due buoni ordini sono sempre l'uno nell'altro;
unendo fra loro tutti i buoni ordini -- o tutte le classe di isomorfismo 
di buoni ordini -- dovrebbe potersi costruire un buon ordine pi\`u grande di tutti. Questa vasta struttura sar\`a inevitabilmente una classe propria:
la classe dei \vocab{numeri ordinali,} i cui elementi sono rappresentanti di tutte le possibili classi di isomorfismo di buoni ordini.

\begin{definition}
	L'insieme $\alpha$ \`e \vocab{transitivo} se $\forall {x \in
\alpha} \; x \subseteq \alpha$, o, equivalentemente, se $\forall {x \in
\alpha} \, \forall {y \in x} \; y \in \alpha$.

{\drkgrn Ossia: diciamo che $\alpha$ \`e transitivo se gli elementi degli
elementi di $\alpha$ sono, a loro volta, elementi di $\alpha$.}
\end{definition}

\begin{definition}[Ordinali di Von Neumann]
	L'insieme $\alpha$ \`e un \vocab{ordinale} se \`e {\drkred
transitivo e bene ordinato dalla relazione di appartenenza.}

Formalmente, l'insieme transitivo $\alpha$
	\`e un ordinale se $(\alpha, <_\alpha)$ \`e un buon ordine, con
	\[ {\drkblu <_\alpha} \;\;\Mydef\;\; \{(x,y) \in \alpha \times \alpha \:|\: x \in y\}
		\]

	Denotiamo con \vocab{$\Ord$} la classe degli ordinali, per cui:
	\[ {\drkblu \alpha \in \Ord} \;\;\Mydef\;\; \text{``$\alpha$ \`e transitivo
e bene ordinato da $\in$''}
		\]
\end{definition}

\begin{example}
\ 

\vskip-\parskip
\vskip-2\baselineskip
\ 
	\begin{itemize}
		\item $\omega$ \`e un ordinale
		\item gli elementi di $\omega$ sono ordinali
		\item $s(\omega) = \omega \cup \{\omega\}$ \`e un ordinale
	\end{itemize}
\end{example}

\begin{remark}
	\label{Ord_trans}
	Se $\alpha \in \Ord$ e $\beta \in \alpha$, allora $\beta \in \Ord$
e $\beta = \alpha_\beta$.\\
	{\drkgrn Nota: in particolare, la classe degli ordinali \`e
transitiva.}
\end{remark}

\begin{proof}
	Siccome $\beta \in \alpha$, per la transitivit\`a di $\alpha$,
$\beta \subseteq \alpha$, quindi $\beta$ \`e un sottoinsieme bene ordinato
dalla restrizione di $<_\alpha$, ovvero da~$\in$.
	La transitivit\`a di $\beta$ segue dalla transitivit\`a della relazione di ordine $<_{\alpha}$. Prendiamo, infatti, $\delta \in \gamma \in \beta$, vogliamo verificare che $\delta \in \beta$, per fare ci\`o osserviamo che per la transitivit\`a di $\alpha$ si ha:
	\[ \gamma \in \beta \in \alpha \implies \gamma \in \alpha \qquad \delta \in \gamma \in \alpha \implies \delta \in \alpha
		\]
	ora abbiamo quindi $\delta,\gamma,\beta \in \alpha$ e sappiamo che $\delta <_\alpha \gamma \land \gamma <_\alpha \beta$, per cui, essendo $<_\alpha$ transitivo, si ottiene $\delta <_\alpha \beta \equiv \delta \in \beta$, per cui $\beta$ \`e transitivo.
	Resta da dire che $\beta = \alpha_\beta$:
	\[ x \in \alpha_\beta \;\;\iff\;\; x \in \alpha \land x <_\alpha
\beta \;\;\iff\;\; x \in \alpha \land x \in \beta
		\]
	Per la transitivit\`a di~$\alpha$ vale $x \in \beta \rightarrow x
\in \alpha$, quindi
	$x \in \alpha \land x \in \beta \iff x \in \beta$ da cui
l'asserto.
\end{proof}

Dimostriamo ora che due ordinali non possono essere nella stessa classe di
isomorfismo.
Vorremmo poi dire che ogni classe di isomorfismo contiene almeno un
ordinale.
Ci servir\`a la seguente osservazione.

\begin{remark}
	Se $f\colon A \to B$ \`e un isomorfismo fra $(A,<_A)$ e $(B,<_B)$, allora preso un qualunque $a \in A$ abbiamo $f[A_a] = B_{f(a)}$.
\end{remark}

\begin{proof}
	Basta semplicemente osservare che:
	\[\begin{split}
		x \in B_{f(a)} &\iff x <_B f(a) \\
					   &\iff f^{-1}(x) <_A a \\
					   &\iff f^{-1}(x) \in A_a \\
					   &\iff x \in f[A_a]
	\end{split}
		\]
	e si conclude per estensionalit\`a.
\end{proof}

\begin{proposition}\label{ord-iso-equ}
	Dati $\alpha,\beta \in \Ord$, se $(\alpha,<_\alpha) \sim (\beta,<_\beta)$, allora $\alpha = \beta$.
\end{proposition}

\begin{proof}
	Sia $f \colon \alpha \to \beta$ un isomorfismo, ci basta
dimostrare che $\forall {\gamma \in \alpha} \; f(\gamma) = \gamma$, cio\`e che $f = \id_\alpha$. Sia, per assurdo,
	$\gamma$ il minimo elemento di $\alpha$ tale che $f(\gamma) \ne \gamma$, allora:
	\[ \gamma = \alpha_\gamma = f[\alpha_\gamma] = \beta_{f(\gamma)} =
f(\gamma)\;\;{\drkred\lightning}
		\]
\end{proof}

Possiamo ora chiederci come si rifletta l'ordinamento totale delle classi di isomorfismo di buoni ordini, dato dalla relazione 
``essere segmento iniziale di'', sugli ordinali. La risposta \`e che diventa la relazione di appartenenza.

\begin{theorem}
	Dati $\alpha,\beta \in\Ord$, vale {\drkred una e una sola} delle seguenti:
	\begin{align*}
		\alpha \in \beta \;\text{che vale se e solo se}\; & (\alpha,<_\alpha) \prec (\beta,<_\beta) \\
		\alpha = \beta \;\phantom{\text{che vale se e solo
se}}\mathllap{\text{---\!---\!--- '' ---\!---\!---}\quad}\; & (\alpha,<_\alpha) \sim (\beta,<_\beta) \\
		\beta \in \alpha \;\phantom{\text{che vale se e solo se}}\mathllap{\text{---\!---\!--- '' ---\!---\!---}\quad}\; & (\beta,<_\beta) \prec (\alpha,<_\alpha)
	\end{align*}
\end{theorem}

{\drkgrn \underline{Notazione}: nella dimostrazione porremo per comodit\`a
$\alpha \prec \beta \Mydef (\alpha,<_\alpha) \prec (\beta,<_\beta)$, e
analogamente per $\alpha \sim \beta$ e $\beta \prec \alpha$.}

\begin{proof}
Per il teorema~\ref{bo-trico-teo}, i tre casi $\alpha\prec\beta$,
$\alpha\sim\beta$ e~$\beta\prec\alpha$ sono alternativi. Resta da
verificare che corrispondono rispettivamente a $\alpha\in\beta$,
$\alpha=\beta$ e~$\beta\in\alpha$.

{\drkblu Se $\alpha \prec \beta$.} Per definizione, $\alpha \sim \beta_\gamma$, per qualche $\gamma \in \beta$, e, per l'\hyperref[Ord_trans]{osservazione}, $\beta_\gamma = \gamma$.
		Per cui $\alpha \sim \gamma$, quindi $\alpha = \gamma$, e,
in particolare, $\alpha \in \beta$. D'altro canto, assumendo
$\alpha\in\beta$, abbiamo $\alpha=\beta_\alpha$, da cui~$\alpha \prec
\beta$.

{\drkblu Se $\alpha \sim \beta$.} Per la proposizione~\ref{ord-iso-equ},
$\alpha = \beta$. Il viceversa \`e immediato.

{\drkblu Il caso $\beta\prec\alpha$} \`e come~$\alpha\prec\beta$.
\end{proof}

\begin{notation}
	Dati $\alpha,\beta \in \Ord$ scriviamo
	\[ {\drkblu \alpha < \beta} \;\;\Mydef\;\; \alpha \in \beta
		\]
	{\drkgrn Il teorema precedente ci dice che la relazione $<$ gode
delle propriet\`a di un ordine totale stretto sulla classe degli
ordinali.}
\end{notation}

\begin{exercise}
	Dimostra che $\alpha$ \`e un ordinale finito se e solo se $\alpha \in \omega$.
\end{exercise}

\begin{proposition}\label{or-le-su}
	Siano $\alpha,\beta \in \Ord$, allora:
	\[ {\drkgrn\qquad\overbrace{\color{black}\alpha \leq \beta}^{\smash{\mathclap{\text{\drkgrn Per def.\
$\alpha<\beta \lor \alpha=\beta$}}}}} \;\;\leftrightarrow\;\; \alpha \subseteq \beta
		\]
\end{proposition}

\begin{proof}
	{\drkblu $\Longrightarrow$} Si danno due casi, se $\alpha =
\beta$, allora $\alpha\subseteq\beta$. Se $\alpha < \beta$, per definizione $\alpha \in \beta$,
		e, poich\'e $\beta$ \`e transitivo, $\alpha \subseteq \beta$.

{\drkblu $\Longleftarrow$} Dato $\alpha \subseteq \beta$, supponiamo per
assurdo che valga $\beta < \alpha$, allora $\beta \in \alpha \subseteq
\beta$, per cui $\beta \in \beta \;{\drkred\lightning}$
(\underline{Esercizio:} perch\'e $\beta\in\beta$ \`e contraddittorio?)
\end{proof}

Ricordiamo che $s(\alpha) \Mydef \alpha \cup \{\alpha\}$. La proposizione seguente ci dice che $s(\alpha)$ \`e, a buon diritto, il successore di $\alpha$, anche quando $\alpha$
\`e un ordinale.

\begin{proposition}
	Dato $\alpha \in \Ord$, $s(\alpha)$ \`e il minimo ordinale $> \alpha$.
\end{proposition}

\begin{proof}
	Occorre inizialmente verificare che $s(\alpha)$ \`e un ordinale.

{\drkblu Transitivit\`a.} Se $\beta \in s(\alpha)$, allora $\beta \in
\alpha$ o $\beta = \alpha$. Nel primo caso, siccome $\alpha$ \`e
transitivo, $\beta\subseteq\alpha\subseteq s(\alpha)$. Nel secondo
$\beta \subseteq \beta \cup \{\beta\} = s(\beta)=s(\alpha)$.

{\drkblu Buon ordine.} Siccome $s(\alpha)$ \`e un insieme di ordinali,
$\in$ \`e un ordine totale su $s(\alpha)$. Dato $X \subseteq s(\alpha)$, con $X \ne \emptyset$,
		abbiamo due casi, o $X = \{\alpha\}$ o $X \cap \alpha \ne
\emptyset$. Nel primo caso $\alpha$ \`e il minimo di~$X$. Nel secondo
caso, $\min(X) = \min(X \cap \alpha)$.

{\drkblu Resta da dimostrare} che $s(\alpha)$ \`e il minimo
ordinale~$>\alpha$. Sia $\alpha < \gamma$, allora $\alpha\in\gamma$ e, per
transitivit\`a, $\alpha\subseteq\gamma$, quindi
$\alpha\cup\{\alpha\}\subseteq\gamma$, vale a dire $s(\alpha)\le\gamma$.
\end{proof}

\begin{corollary}
	$\forall {\alpha,\beta \in\Ord} \; \beta \leq \alpha \leftrightarrow \beta < s(\alpha)$.
\end{corollary}

\begin{proposition}
	Dato un insieme di ordinali $X$:
	\begin{enumerate}[1.]
		\item Se $X \ne \emptyset$, allora esiste il minimo di $X$, detto \vocab{$\min X$}, inoltre $\min X = \bigcap X$.
		\item Esiste il minimo dei maggioranti di $X$, detto
\vocab{$\sup X$}, inoltre $\sup X = \union X$.
		\item C'\`e almeno un ordinale che non appartiene a $X$.
	\end{enumerate}
\end{proposition}

\begin{proof}[Dimostrazione di 1.]
Dimostriamo innanzitutto che il minimo esiste.
Fissiamo $\alpha \in X$. Consideriamo $\mu = \min_{<_{s(\alpha)}}(X \cap s(\alpha))$.
		Questo esiste in quanto $\alpha \in X \cap s(\alpha)$, per
cui questo inisieme non \`e vuoto. 
Osserviamo ora che $\mu = \min X$, infatti, preso $\beta \in X$, o
$\mu\le\alpha<\beta$, oppure $\beta\le\alpha$ da cui $\beta\in X\cap
s(\alpha)$.

Ora verifichiamo che $\mu=\bigcap X$. Per la proposizione~\ref{or-le-su}, $\forall {\gamma
\in X} \; \mu \subseteq \gamma$, quindi $\mu \subseteq \bigcap X$.
		D'altro canto $\mu \in X$, quindi $\bigcap X \subseteq
\mu$.
\end{proof}

\begin{proof}[Dimostrazione di 2.]
\item Dimostriamo che $\union X$ \`e un ordinale.

{\drkblu Transitivit\`a.} Dato $\alpha \in \union X$, esiste $\beta \in
X$ tale che $\alpha \in \beta$. Per la transitivit\`a di $\beta$ si ha
$\alpha\subseteq \beta$, da cui $\alpha\subseteq \union X$.
	
{\drkblu Buon ordine.} Ogni $\alpha \in \union X$ appartiene a qualche
ordinale $\beta \in X$, pertanto \`e un ordinale. Quindi $\union X$ \`e
un insieme di ordinali, ed \`e bene ordinato per il punto 1. della proposizione.

Dimostriamo ora che $\sigma\in\Ord$ \`e un maggiorante di $X$ se e solo
se $\union X \le \sigma$.
\[ \forall {\alpha \in X}\; \alpha\le\sigma \;\;\Leftrightarrow\;\;
\forall {\alpha\in X}\; \alpha\subseteq\sigma \;\;\Leftrightarrow\;\;
\union X\subseteq \sigma \;\;\Leftrightarrow\;\; \union X \le \sigma \]
\end{proof}

\begin{proof}[Dimostrazione di 3.]
Basta considerare $s(\sup X)$.
\end{proof}

\begin{corollary}
Ogni classe di ordinali non vuota ha un minimo elemeto.
\end{corollary}

\begin{proof}
	Consideriamo una classe di ordinali~$C$ e fissiamo $\alpha\in C$, allora $S = s(\alpha)\cap C$ \`e un insieme,
e non \`e vuoto perch\'e contiene $\alpha$. Detto $\mu=\min S$, questo
deve essere anche il minimo di~$C$, perch\'e, preso $\beta\in C$, o
$\beta\le\alpha$, quindi $\beta\in S$, per cui $\mu\le\beta$. Oppure
$\mu\le\alpha<\beta$.
\end{proof}

\begin{corollary}
	Un insieme di ordinali \`e un ordinale se e solo se \`e transitivo.
\end{corollary}

\begin{proof}
	Il buon ordinamento vale per il punto 2. della proposizione.
\end{proof}

\begin{corollary}[Paradosso di Burali-Forti]
	$\Ord$ \`e una classe propria.\\
{\drkgrn Ossia non esiste l'insieme di tutti gli ordinali.}
\end{corollary}

\begin{proof}
	Per il punto 3.\ della proposizione, se $\Ord$ fosse un insieme,
esisterebbe un ordinale che non vi appartiene.
\end{proof}

\begin{breakingnote}[Cosa c'\`e di paradossale nel paradosso di Burali-Forti?]
	Nel 1897, Cesare Burali-Forti era assolutamente convinto della esistenza dell'insieme
	di tutti gli ordinali -- definiti allora come le classi di
isomorfismo di buoni ordini -- quello che non sapeva \`e se la relazione
$\prec$ fosse un ordine totale.
	\begin{figure}[H]
		\centering
		\includegraphics[width = .8\textwidth]{immagini/Burali_Forti.png}
	\end{figure}
	Burali-Forti credette di poter negare la totalit\`a dell'ordine
$\prec$ ragionando per assurdo. Se $\prec$ fosse un ordine totale, \`e
facile vedere che sarebbe buono, ma 
	allora la classe di isomorfismo~$\Omega$ di $(\Ord,\prec)$,
	sarebbe, a sua volta, uno dei membri di $\Ord$. Considerando il suo
successore $s(\Omega)$, avremmo $\Omega \prec s(\Omega)$, ma anche
$s(\Omega) \prec \Omega$ perch\'e $s(\Omega) \in \Ord\;\drkred\lightning.$

	Il guaio \`e che, nello stesso anno, Cantor pubblic\`o una
dimostrazione del fatto che la relazione $\prec$ \`e totale -- esattamente
l'argomento dei segmenti iniziali isomorfi nella dimostrazione del
teorema~\ref{bo-trico-teo}. Come \`e stata risolta la
contraddizione? Concludendo che l'insieme di tutti gli ordinali non esiste?
{\drkred No!} Sfortunatamente, Burali-Forti 
	aveva capito male la definizione di buon ordine. Anche cos\`i, forse, nessuno se ne sarebbe accorto, ma, quel che \`e peggio, aveva tentato di correggerla, facendo in realt\`a un pasticcio.
	La contraddizione \`e stata quindi imputata, da Burali-Forti
stesso e da Cantor, al bisticcio di definizioni, e il paradosso \`e stato dimenticato. Cinque anni dopo, Russell
	not\`o che l'assurdo sussiste anche se si usa la definizione corretta, e fu cos\`i che il paradosso di Burali-Forti acquis\`i il suo nome. E tutti vissero felici e contenti.
\end{breakingnote}

\subsection{L'assioma del rimpiazzamento}
Gli ordinali di Von Neumann sono eleganti, ma quanti ne abbiamo? Si pu\`o
dimostrare che, assumendo i soli assiomi 1--7, il gran totale degli ordinali potrebbe essere:
\[ \textcolor{purple}{\Ord \;\;\overset{?}{=}}\;\; {\drkgrn
\underbrace{\color{purple}\{\emptyset,
s(\emptyset),\ldots,s^n(\emptyset),\ldots,\omega,s(\omega),\ldots,s^n(\omega),\ldots\}}_{\text{{in realt\`a, questo si chiamer\`a $\omega + \omega$}}}}
	\]
ossia la classe degli ordinali raggiungibili a partire da $\emptyset$ o da $\omega$ con un numero finito di applicazioni della mappa successore.

\begin{exercise}
	Dimostra che la classe descritta sopra \`e effettivamente una classe, ossia \`e definita da una formula.
\end{exercise}

Se vogliamo poter rispondere alla domanda ``quanti ordinali esistono?'' occorre un nuovo assioma: l'assioma del rimpiazzamento. Sotto questa ipotesi addizionale, la risposta sar\`a
``tutti quelli che potrebbero esistere'', ossia avremo un ordinale per ogni classe di isomorfismo di buoni ordini. Per formulare l'assioma, ci avvarremo dell concetto di funzione classe.

\begin{definition}
	Date due classi $A$ e $B$ una \vocab{funzione classe} da $A$ a $B$ \`e una formula insiemistica $\varphi(x,y)$ tale che:
	\[ \forall {x \in A} \;\existrun {y \in B} \; \varphi(x,y) \]
{\drkgrn Ossia, una funzione classe \`e una propriet\`a, espressa nel
linguaggio della teoria degli insiemi, che ad ogni $x \in A$ associa un
{\drkred unico} $y \in B$.}
\end{definition}

\begin{notation}[Funzione classe]
	Possiamo denotare una funzione classe $\varphi(x,y)$ da $A$ a $B$
mediante la notazione:
	\[ \drkblu F \colon A \to B \]
	In questo caso, la scrittura $y = F(x)$ \`e una semplice abbreviazione:
	\[ {\drkblu y = F(x)} \;\;\Mydef\;\; y \in B \land \varphi(x,y)
		\]
\end{notation}

\begin{example}[Esempi di funzioni classe]
	Le seguenti sono funzioni classe $V \to V$:
	\[ F_1(x) = x \qquad F_2(x) = \{x\} \qquad F_3(x) = \ps(x) \qquad F_4(x) = s(x)
		\]
\end{example}

\begin{axiom}[Assioma del rimpiazzamento]
	\label{ax8}
	Sia $F \colon V \to V$ una funzione classe, e sia $A$ \`e un
insieme\\
allora la classe ${\drkblu F[A]} \;\Mydef\; \{F(x) | x \in A\}$ \`e un insieme.
		\[ \forall A \; \exists B \; \forall y \; y \in B
\leftrightarrow \exists {x \in A} \; y = F(x)
		\]
\end{axiom}

\begin{remark}
Come per l'\href{ax3}{assioma di separazione} -- nota~\ref{se-sche}, anche questo \`e uno schema di
assiomi: uno per ogni possibile funzione classe $F$.
\end{remark}

\begin{proposition}
	Data una funzione classe $F\colon V \to V$
	\[ \forall A \; \existrun B \; \forall y \; y \in B
\leftrightarrow \exists {x \in A} \; y = F(x)
		\]
quindi l'insieme $f[A]$ \`e univocamente determinato.
\end{proposition}

\begin{proof}
	Estensionalit\`a.
\end{proof}

\begin{remark}
	Dato un insieme $A$ e una funzione classe $G \colon {\drkred A} \rightarrow V$, esiste ed \`e unico l'insieme $G[A]$ tale che:
	\[ \forall y \; y \in G[A] \leftrightarrow \exists {x \in A} \; y = G(x)
		\]
\end{remark}

\begin{proof}
	Ci basta semplicemente applicare l'\hyperref[ax8]{assioma del
rimpiazzamento} alla funzione classe $F \colon V \rightarrow V$ definita come:
	\[ y = F(x) \;\;\Mydef\;\; (x \in A \land y = G(x)) \lor (x \not \in A \land y = \emptyset)
		\]
	ossia:
	\[ F(x) \;\;\Mydef\;\; \begin{cases}
		G(x) &\text{se $x \in A$} \\
		\emptyset &\text{altrimenti}
	\end{cases}
		\]
	infatti $x \in A$ implica $G(x) = F(x)$, per cui $G[A] = F[A]$.
\end{proof}

\begin{exercise}
	Dimostra che, dati due insiemi $A$ e $B$, esiste il loro prodotto cartesiano $A \times B$, usando l'assioma del rimpiazzamento ma senza usare l'assioma delle parti.
\end{exercise}

\begin{theorem}
	Dato un buon ordine $(A,<)$, esiste un unico ordinale $\alpha$ tale che $(A,<)\sim\alpha$.
\end{theorem}

\begin{proof}
	L'unicit\`a segue pech\'e $\alpha \sim \alpha' \rightarrow \alpha
= \alpha'$. Resta quindi da dimostrare l'esistenza di $\alpha$. Sia
	\[ A' \;\;=\;\; \{x \in A | \exists\begingroup\gamma \in
\Ord\endgroup \; A_x \sim \gamma\}
		\]
	Consideriamo la funzione classe $F \colon A' \to \Ord$ data da
	\[ F(x) \;\;=\;\; \text{l'unico $\gamma \in \Ord$ tale che $A_x \sim \gamma$}
		\]
	l'unicit\`a segue perch\'e se $A_x\sim\gamma$ e $A_x\sim\gamma'$,
allora $\gamma\sim\gamma'$ quindi $\gamma=\gamma'$.
	Vogliamo dimostrare che l'insieme $\drkblu \alpha = F[A'] \sim (A,<)$ --
che esiste per rimpiazzamento -- \`e un ordinale isomorfo a $(A,<)$.
	Dimostreremo che $\alpha$ \`e un ordinale, $A'$ \`e un segmento iniziale di $A$, $\alpha \sim A'$, e infine che $A' = A$.

{\drkblu $\alpha$ \`e un ordinale.} Siccome $\alpha$ \`e un insieme di
ordinali, ci basta dimostrare la transitivit\`a.
		Sia $\beta \in \gamma \in \alpha$. Fissiamo un isomorfismo
$f \colon \gamma \to A_x$, per un opportuno segmento iniziale $A_x$
di~$A$. Allora $f_{|\beta} \colon \beta \to f[A_\beta] =
A_{f(\beta)}$ \`e un isomorfismo che testimonia che $f(\beta)\in A'$ e
$\beta=F(f(\beta))\in\alpha$.

{\drkblu $A'$ \`e segmento iniziale di $A$.} Sia $y < x \in A'$. Allora
esiste un isomorfismo $f \colon A_x \to \gamma$, per un
opportuno~$\gamma$. La restrizione $f_{|A_y}\colon A_y \to
\gamma_{f(y)}=f(y)$
\`e un isomorfismo, quindi $y\in A'$.

{\drkblu $A' \sim \alpha$} Sia $f \colon A' \to \alpha$ la funzione
definita da $f(x)=F(x)$, che esiste per l'assioma di separazione.
Dimostriamo che $f$ \`e un isomorfismo di ordini.
		La surgettivit\`a \`e immediata perch\'e, per costruzione,
abbiamo che $\alpha = F[A'] = f[A']$. La monotonia segue da
		\[ x < y \;\;\implies\;\; A_x \prec A_y \;\;\implies\;\; f(x) \prec f(y)
\;\;\implies\;\; x < y
			\]

{\drkblu $A' = A$} Se cos\`i non fosse, per il punto 2., $A'$ sarebbe un segmento iniziale
proprio di~$A$, quindi $A' = A_k$ per un opportuno $k \in A$. Ma allora
$A_k \sim \alpha \in \Ord$ per i punti 1.\ e 3., per cui $k \in A' = A_k
\;{\drkred\lightning}$.
\end{proof}

Una conseguenza del risultato precedente \`e che possiamo definire le
operazioni sugli ordinali come semplice riflesso di quelle sui buoni
ordini.

\begin{definition}[Operazioni sugli ordinali -- v.1]
	Dati $\alpha,\beta \in\Ord$, definiamo \vocab{$\alpha + \beta$}, \vocab{$\alpha\cdot\beta$}, \vocab{$\alpha^\beta$} come, rispettivamente, l'unico ordinale tale che:
	\[ {\drkblu\alpha + \beta} \;\sim\;
(\alpha,<_\alpha)+(\beta,<_\beta) \qquad {\drkblu\alpha \cdot \beta}
\;\sim\; (\alpha,<_\alpha) \cdot (\beta,<_\beta)
		\]\[ {\drkblu\alpha^\beta} \;\sim\; (\alpha,<_\alpha)^{(\beta,<_\beta)}
			\]
\end{definition}

\begin{exercise}
	Dimostra che l'insieme introdotto all'inizio della sezione \`e effettivamente $\omega + \omega$, ossia, pi\`u precisamente:
	\[ \forall x \;\; x \in \omega + \omega \;\; \leftrightarrow
\;\;(\exists {m \in \omega} \; x = m) \lor (\exists {n \in \omega} \; x = \omega +n)
		\]
\end{exercise}

\subsection{Induzione e ricorsione transfinite}

Il piatto forte di questa sezione \`e una seconda applicazione dell'assioma del rimpiazzamento: il teorema di ricorsione transfinita. Questo risultato sar\`a pi\`u chiaro a chi ha, in precedenza, risolto il seguente esercizio.

\begin{exercise}
	Dimostra che esiste un insieme $A$ tale che:
	\[ \forall x \; x \in A \leftrightarrow x = \emptyset \lor \exists y \in A \; x = \{y\}
		\]
	{\drkgrn Ossia, in sostanza, dimostra che esiste:}
	\[
{\drkgrn\big\oldparen\emptyset,\,\{\emptyset\},\,\{\{\emptyset\}\},\,\{\{\{\emptyset\}\}\},\,\ldots\big\}}
		\]
\end{exercise}

L'idea per risolvere questo esercizio \`e contenuta nella dimostrazione del teorema di ricorsione numerabile. Attenzione, per\`o,
che questo teorema non si pu\`o applicare direttamente alla situazione dell'esercizio.

\begin{soln}
	Vorremmo definire la funzione classe $F \colon \omega \to V$ data da:
	\[ F(n) \;\;=\;\; \text{singoletto $n$ volte di $\emptyset$}
		\]
	fatto questo, non sar\`a difficile verificare che $A = F[\omega]$ soddisfa la richiesta. Occorre quindi dimostrare che la definizione informale di $F$ data sopra \`e esprimibile nel linguaggio formale della teoria degli insiemi.

Consideriamo le formule seguenti, che ci interessano per $n\in\omega$
\begin{align*}
	{\drkblu \phi(n,f)} \;\;&\Mydef\;\; \begin{cases}
		\phantom{\land}\;\text{$f$ \`e una funzione} \\
		\land\; \Dom(f) = n+1 \\
		\land\; f(0) = \emptyset \\
		\land\; \forall {i \in n} \; f(i+1) =
\big\oldparen f(i)\big\} 
	\end{cases}\\
	{\drkblu y = F(n)} \;\;&\Mydef\;\; \exists f\;\phi(n,f)\land
y=f(n)
\end{align*}
Per dire che la $F$ cos\`i definita \`e una funzione classe, ci
basta dimostrare che
\[ {\drkred \forall{n\in\omega}\;\existsun f\;\phi(n,f)} \]

{\drkblu Esistenza.} Procediamo per induzione. Il caso $n = 0$ \`e
immediato: basta considerare $f = \{(0,\emptyset)\}$.
		Nel caso $n = m+1$, per ipotesi induttiva, abbiamo una
funzione $g$ tale che $\phi(m,g)$. \`E facile verificare che
$f=g\cup\{(n,\{m\})\}$ soddisfa~$\phi(n,f)$.

{\drkblu Unicit\`a.} Date $f_1$ e $f_2$ che soddisfano
$\phi(n,f_1)$ e $\phi(n,f_2)$, dimostriamo per induzione su~$i$ che
$\forall {i \in \omega} \; i \leq n \to f_1(i) = f_2(i)$. Il caso $i = 0$
\`e immediato perch\'e $f_1(0) = \emptyset = f_2(0)$. Per il caso $i =
j+1 \leq n$, abbiamo l'ipotesi induttiva $f_1(j)=f_2(j)$, quindi
\[f_1(i)
= \{f_1(j)\} = \{f_2(j)\} = f_2(i)\]

	Ora abbiamo la funzione classe $F$. \`E immediato verificare che
$F(0) = \emptyset$. Inoltre, dato $n \in \omega$, $F(n+1) = \{F(n)\}$,
infatti, detta $f$ l'unica funzione che soddisfa $\phi(n+1,f)$, \`e
facile verificare che vale $\phi(n,f_{|n+1})$. Per cui
	\[ F(n+1) = f(n+1) = \{f(n)\} = \{f_{|n+1}(n)\}= \{F(n)\}
		\]

Possiamo finalmente dimostrare che $A=F[\omega]$ soddisfa la richiesta
dell'esercizio. Infatti $\emptyset = F(0) \in A$, e, preso $y \in A = F[\omega]$
con $y\neq\emptyset$, abbiamo $y = F(n)$, con $n \ne 0$, per cui $n = m+1$
	per qualche $m \in \omega$, allora, detto $x = F(m) \in A$, si ha
\[y = F(m+1) = \{F(m)\} = \{x\}\]
\end{soln}

\begin{proposition}[Induzione transfinita -- v.1]
	\label{induz_transf1}
	Data una formula insiemistica $\varphi(x)$.\\
 Se vale $\drkred \forall {\alpha \in \Ord} \;(\forall {\beta < \alpha} \;
\varphi(\beta)) \rightarrow \varphi(\alpha)$,\\
allora vale $\drkblu \forall {\alpha \in \Ord} \; \varphi(\alpha)$.
\end{proposition}

\begin{proof}
	Procedendo per assurdo, consideriamo il minimo $\alpha$ tale
che~$\neg\phi(\alpha)$. Allora $\neg \forall {\beta < \alpha} \;
\phi(\beta)$, ossia $\exists {\beta < \alpha}\; \neg\phi(\beta)$, contro la
minimalit\`a di~$\alpha$.
\end{proof}

\begin{note}[L'induzione transfinita \`e uno schema di teoremi]
	Il principio di induzione transfinita non \`e, letteralmente, un
teorema della teoria degli insiemi, quanto piuttosto uno scherma di
teoremi che ci permette 
	di costruire un diverso teorema per ogni possibile formula
$\varphi$.
\end{note}

C'\`e una chiara analogia fra la forma precedente del principio di induzione transfinita e la forma forte dell'induzione aritmetica.
A volte, per\`o, \`e comodo esprimere l'induzione transfinita in una forma che meglio ricorda il principio di induzione di Peano.

\begin{definition}
Un ordinale $\alpha$ \`e detto \vocab{ordinale successore}
se $\exists \begingroup\beta \in \Ord\endgroup \; \alpha = s(\beta)$. Un ordinale
$\alpha\;{\drkred > 0}$ che non \`e successore \`e detto \vocab{ordinale limite.}
\end{definition}

\begin{remark}
	Un ordinale $\alpha$ \`e successore se e solo se ha un massimo.
\end{remark}

\begin{proof}
	Basta osservare che $\beta$ \`e il massimo di~$\alpha$ se e solo
se $\alpha = s(\beta)$.
\end{proof}

\begin{proposition}[Induzione transfinita -- v.2]
	\label{induz_transf2}
	Sia $\varphi(x)$ una formula insiemistica, se vale:
	\begin{enumerate}[(i)]
		\item $\varphi(0)$ \kern1em\vocab{caso base}
		\item $\forall {\gamma \in \Ord} \; \varphi(\gamma)
\rightarrow \varphi(s(\gamma))$ \kern1em\vocab{caso successore}
		\item Per ogni ordinale limite $\lambda$ si ha $(\forall
{\beta < \lambda} \; \varphi(\beta)) \rightarrow \varphi(\lambda)$ \kern1em\vocab{caso limite}
	\end{enumerate}
	allora $\forall {\alpha \in \Ord} \; \varphi(\alpha)$.
\end{proposition}

\begin{proof}
	Basta verificare l'ipotesi della v.1, ossia che per ogni~$\alpha$
	\[ (\forall {\beta < \alpha} \; \varphi(\beta)) \rightarrow \varphi(\alpha)
		\]
	Se $\alpha$ \`e limite o $0$ abbiamo questa formula tout court.
	Se $\alpha = s(\gamma)$ abbiamo
	\[ \forall {\beta < \alpha} \; \varphi(\beta) \;\iff\;
	   \forall {\beta < s(\gamma)} \; \varphi(\beta) \;\implies\;
\varphi(\gamma) \;\implies\; \varphi(s(\gamma)) \;\iff\;
\varphi(\alpha)
		\]
\end{proof}
