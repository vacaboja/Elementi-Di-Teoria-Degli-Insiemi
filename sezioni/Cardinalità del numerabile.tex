\section{La cardinalit\`a del numerabile -- $\aleph_0$}
\ 

\begin{definition}
	Diciamo che $A$ \`e \vocab{al pi\`u numerabile}, in simboli
$\drkblu |A|\leq\aleph_0$, se $|A| \leq |\omega|$.\\
	Diciamo che $A$ \`e \vocab{numerabile}, in simboli $\drkblu
|A|=\aleph_0$, se $|A| = |\omega|$.\\
	{\drkgrn Il simbolo $\aleph_0$ -- alef con zero -- \`e
semplicemente un'abbreviazione per la sequenza di simboli~$|\omega|$.}
\end{definition}

\begin{remark}
	In altri termini, dire che $A$ \`e al pi\`u numerabile significa dire che c'\`e una funzione iniettiva $A \hookrightarrow \omega$. Dire che \`e numerabile significa dire che c'\`e una bigezione con $\omega$.
\end{remark}

\begin{proposition}\label{dic-num}
	Se $A$ \`e al pi\`u numerabile, allora o $A$ \`e finito o $A$ \`e
numerabile.\\
{\drkgrn Ossia $|A| < \aleph_0$ se e solo se $A$ \`e finito.}
\end{proposition}

Non \`e difficile dimostrare la proposizione direttamente, ma ci conviene, invece, passare attraverso alcune considerazioni che saranno utili in seguito.

In generale, per costruire una bigezione fra due insiemi $A$ e $B$ --
ossia per dimostrare $|A| = |B|$ -- occorre appoggiarsi a qualche struttura definita sugli insiemi $A$
e $B$. Per esempio, abbiamo gi\`a sfruttato la presenza di una funzione successore.
In questo corso, giocheranno un ruolo importante, in questa direzione, le
relazioni d'ordine, e, in particolare -- l'idea \`e di Cantor -- i 
buoni ordini. Ricordiamo che un insieme totalmente ordinato \`e
\vocab{bene ordinato}
(definizione~\ref{def-ben-ord}) se ogni suo sottoinsieme non vuoto ha un
minimo elemento.

Il trucco \`e che un isomorfismo di ordini \`e, in particolare, una
bigezione, e spesso, per costruire bigezioni, fa comodo costruire isomorfismi di ordini.

\begin{definition}[Isomorfismo di ordini]
	Due insiemi (parzialmente) ordinati
	$(A,<_A)$ e $(B,<_B)$ sono \vocab{isomorfi}, in simboli $(A,<_A)
\sim (B,<_B)$ se esiste una bigezione $f\colon A \to B$ tale che:
	\[ \forall {x,y \in A} \; x <_A y \; \leftrightarrow \; f(x) <_B f(y)
		\]
\end{definition}

\begin{remark}
	Due insiemi {\drkred totalmente} ordinati $(A,<_A)$ e $(B,<_B)$
sono isomorfi se e solo se esiste una funzione $f\colon A \to B$
surgettiva e \vocab{strettamente crescente} -- cio\`e tale che:
	\[ \forall {x,y \in A} \; x <_A y \; \rightarrow \; f(x) <_B f(y)
		\]
\end{remark}

\begin{exercise}
	Dimostrare la proposizione enunciata sopra.
\end{exercise}

\begin{remark}\label{ord-fin}
	Sia $(A,<_A)$ totalmente ordinato con $|A| = n \in \omega$. Allora
$(A,<_A) \sim (n,<)$, dove $<$ denota l'ordinamento di $\omega$,
ossia~$\in$.
\end{remark}

\begin{proof}
	Procediamo per induzione su $n$.

{\drkred Caso $n = 0$.} $A = \emptyset$, quindi $(A,<_A) \sim (\emptyset,\emptyset)$ tramite la funzione vuota.

{\drkred Caso $n = m + 1$.} Osserviamo, per iniziare, che $A$ ha un
massimo elemento~$M$. Possiamo, infatti, scrivere $A=A'\cup\{a\}$ con
$|A'|=m$. Per ipotesi induttiva, $A'$, con l'ordine~$<_A$, \`e ismorfo
a~$(m,<)$, quindi ha un massimo elemento~$a'$. Si verifica immediatamente
che il massimo di~$A$ \`e il maggiore fra $a$ e~$a'$.

Poniamo $A'' = A\setminus\{M\}$. Allora $|A''| = m$, per cui, per ipotesi
induttiva, esiste un
isomorfismo di ordini $f\colon A''\to m$ fra $(A'',<_A)$ e~$(m,<)$. Ora
\`e facile verificare che il seguente \`e un isomorfismo di ordini
\begin{align*}
g\colon A &\to m+1 \\
x &\mapsto \begin{cases}
f(x) &\text{se $x <_A M$} \\
m &\text{se $x = M$}
\end{cases}
\end{align*}
\vskip-2\baselineskip
\end{proof}

Possiamo caratterizzare $\omega$ in termini delle propriet\`a del suo
ordinamento naturale. Le propriet\`a chiave sono le seguenti.

\begin{proposition}
\ \\[-2\baselineskip]
	\begin{enumerate}[(1)]
		\item $(\omega,<)$ \`e un buon ordine.
		\item $(\omega,<)$ \`e \vocab{illimitato} -- ossia
$\forall {x \in \omega} \; \exists {y \in \omega} \; x < y$.
		\item Ogni $A \subseteq \omega$ superiormente limitato e non vuoto ha un massimo, ossia:
		\[ \forall {A \subseteq \omega} \; (A \ne \emptyset \land
(\exists {L \in \omega} \; \forall {x \in A} \; x \leq L)) \rightarrow
(\exists {M \in A} \; \forall {x \in A} \; x \leq M)
			\]
	\end{enumerate}
\end{proposition}

\begin{proof}
	(1) \`e il principio del minimo. (2) segue da $x<s(x)$.
Dimostriamo~(3). Se $A$ \`e superiormente limitato da $L \in \omega$,
allora $A \subseteq s(L)$, quindi $A$ \`e finito.
Di conseguenza, applicando l'osservazione~\ref{ord-fin}, si vede che $A$
ha un massimo. {\drkgrn Basta, per esempio, considerare l'ordinamento su~$A$
definito da~$x\prec y \Mydef x > y$, questo \`e isomorfo a un numero
naturale~$n$, l'elemento corrispondente a $0\in n$ \`e il massimo di~$A$.}
\end{proof}

\begin{proposition}\label{car-ome}
	Sia $(A,\prec)$, con $A \ne \emptyset$, un ordinamento:
	\begin{enumerate}[(1)]
		\item buono
		\item illimitato
		\item tale che ogni sottoinsieme superiormente limitato e non vuoto di $A$ ha un massimo secondo $\prec$
	\end{enumerate}
	allora $(A,\prec) \sim (\omega,<)$.
\end{proposition}

\begin{proof}%[Dimostrazione di~\ref{car-ome}]
	Costruiamo per ricorsione un isomorfismo~$f$ da $(\omega, <)$ a
$(A,\prec)$.
	\[ f(x) = \operatorname{min}_\prec \big(A\setminus f[x]\big) \]
dove $\operatorname{min}_\prec$ denota il minimo secondo~$\prec$.

Verifichiamo che $f\colon \omega \to A$ \`e ben definita, strettamente crescente e surgettiva.

{\drkblu Buona definizione $f$.} Dobbiamo verificare che il minimo esista,
ossia che $A\setminus f[x]\neq\emptyset$. Se cos\`i non fosse, $A$ sarebbe
l'insieme finito~$f[x]$, ma l'osservazione~\ref{ord-fin} implica che un
insieme finito e non vuoto non pu\`o essere illimitato.

{\drkblu Monotonia stretta.} Se $x_1<x_2$, allora $f[x_1]\subseteq
f[x_2]$, quindi $A\setminus f[x_1] \supseteq A\setminus f[x_2]$, e, di
conseguenza, $f(x_1) \le f(x_2)$. D'altro canto, per costruzione $f(x_2)
\notin f[x_2] \ni f(x_1)$, quindi $f(x_1)\neq f(x_2)$.

{\drkblu Surgettivit\`a.} Dato $y \in A$ cerchiamo $x \in \omega$ tale che
$f(x) = y$. Osserviamo che c'\`e qualche $x\in\omega$ tale che $y \preceq f(x)$,
altrimenti $f[\omega]$ sarebbe superiormente
limitato da~$y$, quindi dovrebbe avere un massimo, contraddicendo la
monotonia stretta di~$f$. Consideriamo quindi il minimo $x\in\omega$ tale
che $y \preceq f(x)$. Ci basta dimostrare che $f(x)\preceq y$. Dato
$x'<x$, per la minimalit\`a di~$x$, abbiamo $f(x')\prec y$, quindi, in
particolare, $f(x')\neq y$. Ne segue che $y\in A\setminus f[x]$, da cui,
per costruzione, $f(x)\preceq y$.
\end{proof}

Tornando alla proposizione iniziale, dobbiamo dimostrare che un insieme
al pi\`u numerabile \`e finito o numerabile.

\begin{proof}[Dimostrazione di \ref{dic-num}]
	Per ipotesi esiste $f\colon A \rightarrow \omega$ iniettiva, per
cui $|A| = |f[A]|$, e, siccome $f[A] \subseteq \omega$,
	ci basta dimostrare che dato $B \subseteq \omega$, o $B$ \`e finito o \`e numerabile.

	Supponiamo quindi $B\subseteq\omega$  infinito, sostengo che $B$,
con l'ordine indotto da~$\omega$,
soddisfa le ipotesi del lemma precedente, ed \`e quindi numerabile. Le
ipotesi (1) e (3) vagono perch\'e ogni sottoinsieme di $B$ \`e, in
particolare, un sottoinsieme di~$\omega$. Resta da verificare che $B$ \`e
illimitato. 
Se, per assurdo, ci fosse $y\in B$ tale che $\forall x\in B\;x<y$, allora
$B \subseteq y$, contrariamente al fatto che $B$ \`e infinito.
\end{proof}

\begin{exercise}
	\label{disuguaglianze_senza_AC2}
	Dimostra che se $|A| \leq \aleph_0$ e $f\colon A \to B$ \`e surgettiva, allora $|B| \leq \aleph_0$.
\end{exercise}

\subsection{Insiemi numerabili in pratica}
Sapere che, se $|A| \leq \aleph_0$, allora o $A$ \`e finito o \`e
numerabile, ci fornisce lo strumento essenziale per dimostrare la numerabilit\`a di molti insiemi concreti. Spesso, infatti,
\`e facile dimostrare che un insieme infinito \`e tale. Rimane poi da gestire un discorso di disuguaglianze per dire che esso \`e al pi\`u numerabile.

Cominciamo quindi con qualche considerazione generale a proposito delle disuguaglianze fra cardinalit\`a.

\begin{remark}
	\label{monot-op-card}
	Dati gli insiemi $A,B,C$ con $|B| \leq |C|$ allora vale:
	\begin{align*}
		|A| + |B| &\leq |A| + |C| & |A|^{|B|} &\leq |A|^{|C|} \\
		|A| \cdot |B| &\leq |A| \cdot |C| & |B|^{|A|} &\leq |C|^{|A|}
	\end{align*}
\end{remark}

Vale a dire che le operazioni sulle cardinalit\`a sono monotone, nel senso delle disuguaglianze larghe.
{\drkred Attenzione per\`o che, in generale, NON sono strettamente monotone!}

\begin{proof}
	Detta $f\colon B \to C$ la funzione iniettiva che testimonia che $|B| \leq |C|$ e detto $B' = f[B]$ abbiamo che $|B| = |B'|$,
	quindi basta dimostrare le disuguaglianze asserite con $B'$ al
posto di~$B$.
Ora, poich\'e $B' \subseteq C$ otteniamo facilmente:
	\begin{alignat*}{4}
		B' \subseteq C \quad&\implies\quad  & (A \times \{0\}) &\cup (B' \times \{1\})  &\;\;&\subseteq\;\; &  (A \times \{0\}) &\cup (C \times \{1\}) \\
				&\implies &|(A \times \{0\}) &\cup (B' \times \{1\})| &&\leq      & |(A \times \{0\}) &\cup (C \times \{1\})| \\
				&\implies & |A| &+ |B'| && \leq & |A| &+ |C|
	\end{alignat*}
	Le altre si ottengono allo stesso modo.
\end{proof}

\begin{remark}
	$|A \cup B| \leq |A| + |B|$.
\end{remark}

\begin{proof}
	Basta osservare che la seguente funzione \`e iniettiva:
	\begin{align*} f\colon A \cup B &\to (A \times \{0\}) \cup (B \times \{1\})\\
	 x & \mapsto \begin{cases}
		(x,0) &\text{se $x \in A$} \\
		(x,1) &\text{altrimenti}
	\end{cases}
	\end{align*}
\vskip-\baselineskip
\end{proof}

Veniamo ora a calcolare le operazioni aritmetiche. Gi\`a sappiamo, per il \hyperref[cantor]{teorema di cantor}, 
che $2^{\aleph_0} > \aleph_0$, per cui mettere un $\aleph_0$ a esponente di qualunque cosa non sia uno $0$ o un $1$ conduce fuori dal numerabile.
Tutto il resto invece no.

\begin{proposition}
	\[\aleph_0 + \aleph_0 \;\;=\;\; \aleph_0 \cdot \aleph_0 \;\;=\;\; \aleph_0^{n} \;\;=\;\; \aleph_0 \qquad \text{con $n \in \omega\setminus\{0\}$.} \]
\end{proposition}

\begin{proof}
\def\qedsymbol{}
	Supponiamo di sapere gi\`a che $\aleph_0 \cdot \aleph_0 = \aleph_0$, allora possiamo formare la catena di disuguaglianze:
	\[ \aleph_0 \;\;=\;\; \aleph_0 + 0 \;\;\leq\;\; \aleph_0 +
\aleph_0 \;\;=\;\; \aleph_0 \cdot 2 \;\;\leq\;\; \aleph_0 \cdot \aleph_0
\;\;=\;\; \aleph_0
		\]
	Da cui per il teorema di \hyperref[CB]{Cantor-Bernstein}
	\[ \aleph_0 + \aleph_0 \;\;=\;\; \aleph_0 \cdot \aleph_0 \;\;=\;\; \aleph_0
		\]
	Ora \`e facile vedere per induzione che $n \in \omega \setminus\{0\} \rightarrow \aleph_0^n = \aleph_0 $, infatti $\aleph_0^1 = \aleph_0$, quindi $\aleph_0^{n+1} = \aleph_0^n \cdot \aleph_0 = \aleph_0 \cdot \aleph_ 0 = \aleph_0$.
\end{proof}

Per concludere la dimostrazione precedente, resta da dimostrare che
$\aleph_0 \cdot \aleph_0 = \aleph_0$.

\subsection{Prodotto di numerabili \`e numerabile}
\ 

\begin{lemma}
	$\aleph_0 \cdot \aleph_0 = \aleph_0$ {\drkgrn ossia esiste una
bigezione fra $\omega \times \omega$ e $\omega$.}
\end{lemma}

{\drkgrn
\begin{wrapfigure}[17]{r}{4.5cm}
\vskip-2\baselineskip
	\includegraphics[width=4.5cm]{immagini/aleph2.png}
\end{wrapfigure}
Ci sono diverse vie per illustrare questo risultato. Per esempio, possiamo rappresentare le coppie $(x,y) \in \omega \times \omega$ sotto la specie di una griglia a maglie quadrate.
Poi disegnare un percorso che pare visitare tutte le maglie della griglia, con sufficiente apparenza di regolarit\`a, possibilmente, da convincere il lettore che vi debba essere un metodo.
Infine numeriamo le maglie secondo l'ordine in cui sono visitate dal percorso. Avremo cos\`i numerato tutte le coppie di numeri naturali del disegno.

Altrimenti, possiamo esibire delle bigezioni esplicite, per esempio:
\[ f(x,y) = 2^x \cdot (2y + 1) - 1 \qquad g(x,y) = \frac{(x+y)^2 + 3x + y}{2}
	\]
\`E possibile scrivere i due numeri della coppia in base 10 a cifre
alterne, tipo: $({\drkblu 64},{\drkred 4096}) \mapsto
{\drkred 4}\textcolor{blue!50!cyan}{0}{\drkred 0}\textcolor{blue!50!cyan}{0}{\drkred 9}{\drkblu 6}{\drkred 6}{\drkblu 4}$.

E saranno possibili mille altre furbit\`u, ma noi useremo, invece, il tipo
d'ordine di~$\omega$.

}

\vfill\pagebreak
\vbox to 0pt{\vskip4\baselineskip\includegraphics[width=.23\textwidth]{immagini/ordine_omega.png}\vss}
\begin{proof}
Innanzitutto fissiamo l'ordine seguente su $\omega \times \omega$:
\begin{align*}
	(a,b) \prec (a',b') \;\;\Mydef\phantom{\lor}\;& \max(a,b) < \max(a',b') \\
						\lor\;& (\max(a,b) = \max(a',b') \land a < a') \\
						\lor\;& (\max(a,b) = \max(a',b') \land a = a' \land b < b')
\end{align*}

\setlength\leftskip{.25\textwidth}
{\drkgrn Ossia per confrontare $(a,b)$ con $(a',b')$, si confrontano prima
$\max(a,b)$ e $\max(a',b')$; a parit\`a si confrontano $a$ ed $a'$; se
anche queste coincidono, allora si confrontano $b$ e $b'$.

L'idea \`e che le coppie $\prec$ di una certa
$(a,b)$ fissata sono tutte contenute nel quadrato $\{0,\ldots,\max(a,b)\}
\times \{0,\ldots,\max(a,b)\}$, per cui sono in numero finito, e questo
implica che $(\omega \times \omega, \prec)$ \`e isomorfo a $(\omega,<)$.}

	Formalmente, iniziamo col verificare che $\prec$ sia effettivamente un ordine stretto e totale.

\setlength\leftskip{0pt}
{\drkblu Irriflessivit\`a.} Immediata.

{\drkblu Transitivit\`a.} Prendiamo $(a,b) \prec (a',b) \prec (a'',b'')$,
per cui
\[\max(a,b) \leq \max(a',b') \leq \max(a'',b'')\]
Se una delle
disuguaglianze \`e stretta, allora $(a,b) \prec (a'',b'')$. Altrimenti
\[\max(a,b) = \max(a',b') = \max(a'',b'')\]
per cui $a \leq a' \leq a''$.
Nuovamente, se una delle disuguaglianze \`e stretta abbiamo concluso.
Altrimenti $a=a'=a''$, da cui segue che $b < b' < b''$, quindi $(a,b) \prec (a'',b'')$.

{\drkblu Totalit\`a.} Se $(a,b)$ ed $(a',b')$ non sono n\'e~$\prec$
n\'e~$\succ$, allora $\max(a,b) = \max(a',b')$, $a = a'$ e $b = b'$,
quindi $(a,b)=(a',b')$.

Ora vogliamo dire che $(\omega \times \omega, \prec) \sim (\omega, <)$.
Questo conclude la dimostrazione perch\'e un isomorfismo \`e, in
particolare, una bigezione.
Ci servir\`a la nozione seguente. Se $(a,b) \in \omega \times \omega$
definiamo
	\[ (\omega \times \omega)_{(a,b)} \;\;\Mydef\;\; \{(x,y) \in \omega \times \omega | (x,y) \prec (a,b)\}
		\]
	{\drkgrn detto ``\vocab{segmento iniziale} determinato da $(a,b)$
su $(\omega \times \omega,\prec)$''.}\\
Osserviamo che $(\omega \times \omega)_{(a,b)}$ \`e finito, perch\'e \`e
un sottoinsieme dell'insieme finito $(\max(a,b)+1) \times (\max(a,b)+1)$.

	Ci serve dire che {\drkred 1.} $(\omega \times \omega, \prec)$ \`e
bene ordinato {\drkred 2.} $(\omega \times \omega, \prec)$ \`e illimitato
{\drkred 3.} ogni sottoinsieme non vuoto e superiormente limitato di $\omega \times \omega$ ha un massimo.

	\begin{enumerate}[1.]
		\item Dato $A \subseteq \omega \times \omega$ con $A \ne
\emptyset$, fissiamo $a \in A$. Sia $A' = (\omega \times \omega)_a \cap
A$. Se $A'=\emptyset$, allora $a$ \`e il minimo di~$A$. Altrimenti, $A'$
\`e finito, non vuoto, e totalmente ordinato da~$\prec$, dunque ha un
minimo~$m$. Ora, $m$ deve essere il minimo di~$A$, perch\'e, se $x \in A
\setminus A'$, allora $m < a \leq x$.
		\item Dato $(a,b) \in \omega \times \omega$, $(a,b) \prec
(s(a),s(b))$.
		\item Dato $A \subseteq \omega \times \omega$ non vuoto e
superiormente limitato da $(a,b) \in \omega \times \omega$, abbiamo che $A
\subseteq (\omega \times \omega)_{(a+1,b+1)}$ \`e finito, quindi ammette
massimo perch\'e $\prec$ \`e totale.
	\end{enumerate}
\end{proof}

\subsection{Numeri interi e razionali}
Usando la proposizione appena dimostrata, potremmo dimostrare, per
esempio, che $\ZZ$ e $\QQ$ sono numerabili, se non fosse che non abbiamo
ancora definito questi oggetti. Allo scopo, ricordiamo che --
esercizio~\ref{3.73} -- una relazione di equivalenza
induce un insieme di classi di equivalenza.

\begin{definition}[$\ZZ$]
	Definiamo $\ZZ$ come l'insieme delle classi di equivalenza su $\omega \times \omega$ indotte dalla relazione
	\[ (a,b) \sim_\ZZ (a',b') \;\;\Mydef\;\; a + b' = b + a' \qquad
\text{\drkgrn morale: $(a,b) = a - b$}
		\]
\end{definition}

\begin{exercise}
	Dimostra che $\sim_\ZZ$ \`e una relazione di equivalenza.
\end{exercise}

\begin{exercise}
	Definiamo $+,-,\cdot$ su $\ZZ$ mediante:
	\begin{align*}
		[(a,b)]_\ZZ + [(a',b')]_\ZZ \;\;&\Mydef\;\; [(a+a',b+b')]_\ZZ \\
		-[(a,b)]_\ZZ \;\;&\Mydef\;\; [(b,a)]_\ZZ \\
		[(a,b)]_\ZZ \cdot [(a',b')]_\ZZ \;\;&\Mydef\;\; [(a\cdot a' + b \cdot b', a \cdot b' + a' \cdot b)]_{\ZZ}
	\end{align*}
	Dimostra che $\ZZ$, con queste operazioni, \`e un anello
commutativo con identit\`a~$1 \,\Mydef\, [(1,0)]_\ZZ$. 
\end{exercise}

\begin{definition}[$\QQ$]
	Definiamo $\QQ$ come l'insieme delle classi di equivalenza su $\ZZ \times (\omega\setminus\{0\})$ indotte dalla relazione:
	\[ (n,d) \sim_\QQ (n',d') \;\;\Mydef\;\; n \cdot d' = n' \cdot d
\qquad \text{\drkgrn morale: $(n,d) = \frac nd$}
		\]
\end{definition}

\begin{exercise}
	Dimostra che $\sim_\QQ$ \`e una relazione di equivalenza.
\end{exercise}

\begin{exercise}
	Definisci $+,-,\cdot$ e $\square^{-1}$ su $\QQ$ nella maniera ragionevole e dimostra che $\QQ$ \`e un campo.
\end{exercise}

\begin{exercise}[Ordine di $\QQ$]
	Definisci la relazione $<$ su $\QQ \times \QQ$ dicendo che $q \in \QQ$ \`e positivo se $q = [(n,d)]_\QQ$, con $n,d \in \omega\setminus\{0\}$, e dicendo che $a < b$ se e solo se $b - a$ \`e positivo.
	Dimostra che questo \`e un ordine totale e \vocab{denso}, cio\`e:
	\[ \forall {a,b \in \QQ} \; a < b \rightarrow \exists {c \in \QQ} \; a< c <b
		\]
\end{exercise}

\begin{note}
	Gli esercizi precedenti sono tediosi, ma non sono difficili. Nel resto del corso daremo per scontate le propriet\`a aritmetiche elementari di $\ZZ$ e $\QQ$.

Per dimostrare la numerabilit\`a di $\ZZ$ e $\QQ$, \`e comodo richiamare
ancora un esercizio~(\ref{disuguaglianze_senza_AC2}), per\`o, questa

\begin{lemma}
	Se $|A| \leq \aleph_0$ e $f\colon A \to B$ \`e surgettiva, allora $|B| \leq \aleph_0$.
\end{lemma}

\begin{proof}
Per ipotesi esiste $g\colon A\to\omega$ iniettiva, dobbiamo trovare 
$h\colon B \to \omega$ iniettiva. Definiamo
        \[ h(b) = \min\big(g\big[\,\{a\in A\,|\,f(a)=b\}\,\big]\big)
                \]
        L'insieme fra graffe non pu\`o essere vuoto per la surgettivit\`a
di~$f$, per cui il minimo \`e ben definito.
        Dimostriamo l'iniettivit\`a. Se $h(b) = h(b')$, allora i minimi,
siano $g(a)$ e $g(a')$, sono uguali, quindi $a=a'$ per l'iniettivit\`a
di~$g$. A questo punto $b=f(a)=f(a')=b'$.
\end{proof}


\begin{corollary}
	\label{disugcardnum}
	Un insieme $A \ne \emptyset$ \`e al pi\`u numerabile se e solo se
esiste $f \colon \omega \to A$ surgettiva.
\end{corollary}

\begin{proof}
	La freccia $\Leftarrow$ \`e il lemma. Supponiamo ora che $A$ sia
al pi\`u numerabile. Allora o $A$ \`e numerabile, e in questo caso
c'\`e una bigezione~$\omega\to A$. O $A$ \`e finito, per cui c'\`e $h\colon
n \to A$, per qualche $n\in\omega$, bigettiva. In questo caso, fissiamo un $a\in A$, e
definiamo $f(x) = h(x)$ per $x<n$ e $f(x)=a$ per $x\ge n$.
\end{proof}

\begin{notation}
	Con \vocab{successione} (numerabile) intendiamo semplicemente una
funzione con dominio $\omega$, e scriviamo $\drkblu \alpha =
\{\alpha_i\}_{i \in \omega}$ per dire che $\alpha$ \`e una successione. In
questo contesto ${\drkblu \alpha_i} \;\Mydef\; \alpha(i)$.

Per \vocab{enumerazione} di $A$
	intendiamo una successione che mappa surgettivamente $\omega$
su~$A$.
\end{notation}

{\drkgrn Il corollario sopra, quindi, dice che $A \ne \emptyset$ \`e al pi\`u numerabile se e solo se ha un'enumerazione.}

\begin{example}
	$\ZZ$ \`e numerabile.
\end{example}

\begin{proof}
	La funzione $\omega \times \omega \to \ZZ,\quad (a,b) \mapsto a - b$ \`e surgettiva
e $\omega \times \omega$ \`e numerabile, quindi $|\ZZ|\leq \aleph_0$.
	D'altro canto, la funzione $\omega \rightarrow \ZZ,\quad n \mapsto
[(n,0)]_\ZZ$ \`e iniettiva, quindi $\aleph_0 \leq |\ZZ|$. Per
il teorema di \hyperref[CB]{Cantor-Bernstein}, $|\ZZ| = \aleph_0$.
\end{proof}

\begin{example}
	$\QQ$ \`e numerabile.
\end{example}

\begin{proof}
	Come nell'esempio precedente, $\ZZ \times (\omega \setminus\{0\})
\rightarrow \QQ\quad  (n,d) \mapsto \frac{n}{d}$ \`e surgettiva, e $|\ZZ
\times (\omega \setminus\{0\})| = |\ZZ| \cdot |\omega \setminus\{0\}| =
\aleph_0 \cdot \aleph_0 = \aleph_0$, dunque $|\QQ|\leq \aleph_0$.
	Viceversa, la funzione $\omega \rightarrow \QQ,\quad n \mapsto
\frac{n}{1}$ \`e iniettiva, per cui $\aleph_0 \leq |\QQ|$, e si conclude
nuovamente per il teorema di \hyperref[CB]{Cantor-Bernstein}.
\end{proof}

Il corollario~\ref{disugcardnum} e il teorema
di~\hyperref[CB]{Cantor-Bernstein} sono utili strumenti per dimostrare la
numerabilit\`a di svariati insiemi. Il nostro prossimo esempio sar\`a
l'insieme dei numeri reali algebrici, e per questo \`e comodo uno
strumento in pi\`u: il fatto che se abbiamo un insieme $A$ al pi\`u numerabile, e tutti i suoi elementi sono, a loro volta, insiemi al pi\`u numerabili, allora
$\bigcup A$ \`e al pi\`u numerabile.
Questo \`e ragionevole: se esiste una enumerazione $\{a_i\}_{i \in
\omega}$ di $A$, e, per ogni $i \in \omega$, esiste una enumerazione di
${\alpha_{i,j}}_{j\in\omega}$,
allora possiamo mandare surgettivamente $\omega \times \omega$ in $\bigcup
A$: $(i,j)\mapsto\alpha_{i,j}$.

{\drkred L'errore} \`e credere di poter fissare una enumerazione
$\alpha_i$ di $a_i$ per ogni $i \in \omega$. Usando l'assioma della scelta
potremo farlo, ma, per ora, non abbiamo modo, in generale, di procurarci
la funzione $i\mapsto\alpha_i$.
Possiamo per\`o assumere di averla, cos\`i si corregge il ragionamento impreciso di prima.

\begin{proposition}
	Sia $A = \{a_i \in A| i \in \omega\}$ e sia $\{\alpha_i\}_{i \in \omega}$ una successione di funzioni tali per cui,
	per ogni $i \in \omega$, la funzione $\alpha_i\colon \omega \to a_i$ \`e una enumerazione di $a_i$. Allora $\left|\bigcup A\right| \leq \aleph_0$.
\end{proposition}

\begin{proof}
	Basta osservare che la funzione:
\[f \colon \omega \times \omega \to \bigcup A \qquad (i,j) \mapsto
\alpha_i(j) \]
	\`e surgettiva.
\end{proof}

\begin{notation}
	Data una funzione $f \colon I \rightarrow S$ definiamo:
	\[ \bigcup_{i \in I} f(i) \;\;\Mydef\;\; \bigcup f[I]
		\]
	Cos\`i, per esempio, se $A = \{a_i | i \in \omega\}$:
	\[ \bigcup_{i \in \omega} a_i \;\;=\;\; \bigcup A \;\;{\drkgrn =\;\; \{x | \exists i \in \omega \; x \in a_i\}}
		\]
\end{notation}

\begin{definition}
	Definiamo le \vocab{parti finite} di un insieme $A$ come:
	\[ \psf(A) \;\;\Mydef\;\; \{X \in \ps(A) | |X| < \aleph_0\}
		\]
\end{definition}

\begin{proposition}
	$|A| \;\leq\; \aleph_0 \;\;\rightarrow\;\; |\psf(A)| \;\leq\; \aleph_0$.
\end{proposition}

\begin{proof}
	Il caso $A = \emptyset$ \`e immediato. Assumiamo $A \ne \emptyset$, sia:
	\[ \ps^{\leq n} = \{X \in \ps(A) : |X| \leq n\}
		\]
	siccome $\psf(A)$ \`e l'unione degli insiemi~$\ps^{\leq n}(A)$, ci
basta esibire una successione di enumerazioni~$\alpha_n$ di~$\ps^{\leq
n}(A)$.

Sfruttando il fatto che $|\omega \times A| = \aleph_0$, fissiamo una
funzione surgettiva $f\colon
\omega \to \omega \times A$. Indichiamo con $f_\omega(x)\in\omega$ e $f_A(x)\in
A$ le due
componenti della coppia $f(x)$.
A questo punto, possiamo costruire la successione di enumerazioni
$(\alpha_n)_{n \in \omega}$ per ricorsione:

{\drkblu Per $n = 0$.} Siccome $\ps^{\le0}(A) = \{\emptyset\}$, stabiliamo
che $\alpha_0$ \`e la funzione costante $i\mapsto\emptyset$.

{\drkblu Per $n = m+1$.} Supponiamo data una enumerazione $\alpha_m$
di~$\ps^{\leq m}(A)$ e poniamo:
		\[ 	\alpha_{m + 1}(x)\;=\; \begin{cases}
			\emptyset &\text{se $x = 0$} \\
			\alpha_m(f_1(x-1)) \cup \{f_2(x - 1)\} &\text{altrimenti}
			\end{cases}
		\]

	Verifichiamo per induzione che, per ogni~$n\in\omega$, la funzione
$\alpha_n$ \`e una surgezione $\omega\to\ps^{\leq n}(A)$.

{\drkred Caso $n = 0$.} Immediato.

{\drkred Caso $n = m + 1$.} Per ipotesi induttiva la successione $\alpha_m
\colon \omega \to \ps^{\leq m}(A)$ \`e surgettiva, vogliamo verificare che
anche $\alpha_{m + 1} \colon \omega \to \ps^{\leq m + 1}(A)$ lo \`e.

Dato $Y \in \ps^{\leq m+1)}(A)$ si danno due casi. Se $Y = \emptyset$,
allora per costruzione $Y = \alpha_{s(m)}(0)$. Altrimenti fissiamo $y \in
Y$, per cui $|Y\setminus\{y\}| \leq m$, quindi c'\`e un $t\in\omega$ per
cui $Y \setminus\{y\} = \alpha_m(t)$.
		Per la surgettivit\`a di $f \colon \omega \to \omega
\times A$, la coppia $(t,y)$ \`e immagine di qualche $x \in \omega$,
quindi:
		\begin{alignat*}{3}
			\alpha_{m + 1}(x+1) \;&=\;\;& \alpha_m(f_1(x))
&\;\cup\;
\{f_2(x)\} && \\
							   &=& (Y
\setminus\{y\}) \;\;&\;\cup\;\;\;\; \{y\} && =\; Y 
		\end{alignat*}
\vskip-\baselineskip
\end{proof}

\paragraph*{Applicazione}

Dimostriamo che l'insieme dei numeri reali algebrici $\A_{\RR}$ \`e
numerabile. Per questa applicazione, assumiamo le propriet\`a algebriche elementari di $\RR$.
L'insieme $\A_{\RR}$ \`e definito come l'insieme degli $x \in \RR$ che sono zeri di qualche polinomio a coefficienti razionali:
\[ \A_{\RR} \;\;\Mydef\;\; \{x \in \RR | \exists p(x) \in \QQ[x]\setminus\{0\} \; p(x) = 0\}
	\]
I numeri reali che non sono algebrici si dicono \vocab{trascendenti}.
Siccome -- dimostreremo questo risultato in seguito -- $\RR$ non \`e
numerabile, si pu\`o dedurre dalla numerabilit\`a di
$\A_{\RR}$ che ci sono numeri reali trascendenti.

Dimostriamo, intanto, che l'insieme $\QQ[x]$, dei polinomi a coefficienti razionali nella indeterminata $x$, \`e numerabile.
I polinomi, come accade per i numeri, le coppie ordinate, le funzioni,
\etc.\ non sono oggetti della teoria degli insiemi, per cui occorre
fissare una codifica. \`E ragionevole identificare un polinomio:
\[ p(x) = a_0 + a_1x + a_2x^2 + \ldots + a_d x^d
	\]
con l'insieme dei suoi monomi:
\[ p(x) = \{a_0,a_1x,a_2x^2,\ldots,a_dx^d\}
	\]
e ciascun monomio con la coppia {\drkblu (grado, coefficiente):}
\[ p(x) = \{(0,a_0),(1,a_1),\ldots,(d,a_d)\} \]
Rappresentando i polinomi in questo modo, abbiamo:
\[ \QQ[x] \subseteq \psf(\omega \times \QQ)
	\]
per cui $|\QQ[x]| \leq \aleph_0$.
Inoltre \`e elementare immergere $\QQ$ in~$\QQ[x]$, quindi~$|\QQ[x]| = \aleph_0$.

Venendo ad $\A_{\RR}$ abbiamo una facile surgezione:
\begin{align*}
	f \colon (\QQ[x]\setminus\{0\}) \times \omega &\to \A_{\RR}\\
	 (p,i) &\mapsto \text{``la $i$-esima radice di $p$ se questa esiste, altrimenti 0''}
\end{align*}
Per i miei 2.5 lettori che non sono convinti del virgolettato,
descriviamo
$f$ con precisione, mediante una formula insiemistica:
\[
	\text{``$\alpha$ \`e la $i$-esima radice di $p$''} \;\;\Mydef\;\;
	p(\alpha) = 0 \land \bigl|\:\{x \in \RR | x \leq \alpha \land p(x)
= 0\}\:\bigr| = i
\]
\begin{align*}
	y = f(p,i) \;\;\Mydef\; \phantom{\lor}\; &\text{``$y$ \`e la $i$-esima radice di $p$''} \\
	\lor\; &(y = 0 \land \neg \exists \alpha \in \RR \; \text{``$\alpha$ \`e la $i$-esima radice di $p$''})
\end{align*}
Dove viene lasciato come {\drkred esercizio} di
scrivere una formula insiemistica che dice $p(x)=y$.
Per separazione esiste, quindi, $f$, e, di conseguenza $|\A_{\RR}| \leq
\aleph_0$. La disuguaglianza opposta \`e immediata perch\'e $\QQ \subseteq
\A_{\RR}$.

\begin{exercise}[Ogni gruppo finitamente generato \`e al pi\`u numerabile]
	Dato un insieme $X$, una funzione $f\colon X \times X \to X$, e un
sottoinsieme $A \subseteq X$ al pi\`u numerabile, dimostra che esiste un
$\ol A \subseteq X$ al pi\`u numerabile tale che $A \subseteq \ol A$ e~$f[\ol A \times \ol A] \subseteq \ol A$.
\end{exercise}

\subsection{Ordini densi numerabili}
Il prossimo risultato \`e, come al solito, dovuto a Cantor, e caratterizza l'ordine di $\QQ$ a meno di isomorfismi.

\begin{definition}[Densit\`a]
	Sia $(A,<)$ totalmente ordinato, e $B \subseteq A$, diciamo che:
\begin{alignat*}{1}
\text{\drkblu $B$ \`e denso in $(A,<)$} \qquad \mathclap{\text{se}} \qquad
&\forall {x,y \in A} \; x < y \rightarrow \exists {z \in B} \; x < z < y \\
\text{\drkblu $(A, <)$ \`e denso} \qquad \mathclap{\text{se}} \qquad &\text{$A$ \`e
denso in s\'e} \\
\mathclap{\text{cio\`e}}\qquad &\forall {x,y \in A} \; x < y \rightarrow \exists {z \in A} \; x < z < y
\end{alignat*}
\end{definition}

\begin{example}
	$\QQ$ \`e denso, infatti: $ x < y \rightarrow x < \frac{x+y}{2} <
y $.
\end{example}

\begin{notexample}
	L'insieme $\omega$, con l'ordine usuale, non \`e
denso, perch\'e $\nexists {z \in \omega} \; 0 < z < 1$.
\end{notexample}

\begin{theorem}[Teorema di isomorfismo di Cantor]
	\label{isoCantor}
	Sia $(A,<)$ un insieme totalmente ordinato. Allora
$(A,<)\sim(\QQ,<)$ se e solo se valgono simultaneamente:
	\begin{enumerate}[1.]
		\item $|A| = \aleph_0$
		\item $(A,<)$ \`e denso
		\item $(A,<)$ non ha \vocab{estremi}, ossia non ha n\'e
massimo n\'e minimo elemento.
	\end{enumerate}
\end{theorem}

{\drkgrn $\QQ$ soddisfa le tre propriet\`a, per cui, la direzione da
dimostrare \`e che se $(A,<)$ soddisfa 1.--3.\ allora \`e isomorfo
a~$\QQ$. L'idea \`e di costruire un isomorfismo per ricorsione. Ad ogni
passo della ricorsione abbiamo un isomorfismo $f_i \colon A_i \rightarrow Q_i$ con $A_i \subseteq A$ finito 
e $Q_i \subseteq \QQ$. Estendiamo questo isomorfismo facendo crescere
gradualmente $A_i$ e~$Q_i$ fino a invadere $A$ e~$\QQ$ rispettivamente.

Consideriamo il caso in cui vogliamo estendere $f_i$ aggiungendo un nuovo
elemento~$x$ al suo dominio, e senza violare la definizione di
isomorfismo. Per questo dobbiamo mappare~$x$ in un razionale~$f_{i+1}(x)$
che sia situato rispetto agli elementi di~$f_i[A_i]$ nello stesso modo in
cui $x$ \`e situato rispetto agli elementi di~$A_i$.
Siccome $A_i$ \`e finito, per sapere la posizione di $x$ rispetto a
ciascuno degli elementi di $A_i$, ci basta sapere quale sia l'ultimo elemento prima di $x$,
e quale sia il primo dopo $x$ -- diciamo che, per esempio, sono $a_2$  e
$a_3$ rispettivamente. Dobbiamo allora mandare $x$ in un $f_{i+1}(x)$ con $f_i(a_2) < f_{i+1}(x) < f_i(a_3)$, e questo esiste per la densit\`a di $\QQ$.

Ragionando simmetricamente, possiamo anche estendere $f_i$, dato un $y \in \QQ$ con $y \not\in Q_i$, in modo tale che $y \in \Imm(f_{i+1})$.

In definitiva, ci basta quindi fissare un'enumerazione di $A$ e una di $\QQ$, e fare questi passi di estensione in maniera alternata, assicurandoci cos\`i di aggiungere al dominio della $f$, uno per uno,
tutti gli elementi di $A$, e di aggiungere all'immagine, uno per uno,
tutti gli elementi di $\QQ$. La chiave \`e che il procedimento sar\`a
completo solo dopo $\omega$ passi, mentre, ad ogni passo, dobbiamo
estendere soltanto un isomorfismo parziale che ha dominio e immagine
finite, ragion per cui troviamo sempre spazio per farlo.}

Ci far\`a comodo la seguente osservazione.

\begin{remark}\label{obs-uni-fun}
	Sia $F \subseteq \ps(A \times B)$ un insieme di funzioni. Se:
	\[ \forall {f_1,f_2 \in F} \; \underbrace{f_{1|\Dom(f_1) \cap
\Dom(f_2)} = f_{2 |\Dom(f_1) \cap \Dom(f_2)}}_{\text{ossia $\forall {x \in
\Dom(f_1) \cap \Dom(f_2)} \; f_1(x) = f_2(x)$}}
			\]
allora $\bigcup F$ \`e una funzione da $\bigcup\{\Dom(f) | f\in
F\}$ a~$B$.
\end{remark}

\begin{proof}
	Bisogna verificare che se $(x,y_1) \in \bigcup F$ e $(x,y_2) \in \bigcup F$, allora $y_1 = y_2$.
Dalla prima abbiamo che esiste $f_1 \in \bigcup F$ tale che $f_1(x) = y_1$
e, dalla seconda, esiste $f_2 \in \bigcup F$ tale che $f_2(x) = y_2$.
Quindi $x\in \Dom(f_1) \cap \Dom(f_2)$ e, usando l'ipotesi, segue che $y_1 = f_1(x) = f_2(x) =
y_2$.
\end{proof}

\begin{proof}[Dimostrazione di~\ref{isoCantor}.]
	Fissiamo enumerazioni di $A$ e $\QQ$ rispettivamente:
	\[ A = \{a_i | i \in \omega\} \qquad \QQ = \{q_i | i \in \omega\}
		\]
	Intendiamo costruire una successione di funzioni $(f_i)_{i \in
\omega}$ tale che, per ogni $i \in \omega$:
	\begin{enumerate}[(1)]
		\item $f_i \colon A_i \to Q_i$ con $|A_i| = |Q_i| < \aleph_0$
		\item $f_i$ \`e un isomorfismo di ordini fra $A_i$ e $Q_i$
		\item $f_i \subseteq f_{i+1}$ {\drkgrn ossia $f_{i+1}$
estende $f_i$.}
		\item $\forall {j < i} \; a_j \in A_i \land q_j \in Q_i$
{\drkgrn ossia $\begin{aligned} \{a_0,\ldots,a_{i-1}\} &\subseteq
\Dom(f_i) = A_i \\
\{q_0,\ldots,q_{i-1}\} &\subseteq \Imm(f_i) = Q_i
\end{aligned}$}
	\end{enumerate}

Assumendo di aver costruito una successione che verifica queste
condizioni, verifichiamo che $f \Mydef \bigcup_{i \in \omega} f_i$ \`e un
isomorfismo di ordini fra~$A$ e~$\QQ$. Da (3) segue, mediante una facile
induzione, che $\forall {i < j} \; f_i \subseteq f_j$, per cui, applicando
l'osservazione~\ref{obs-uni-fun}, $f$ \`e una funzione. Inoltre (4)
implica che $\Dom(f)=A$ e~$\Imm(f)=\QQ$. Fissiamo $x,y\in A$ e
verifichiamo
\[x < y \;\;\leftrightarrow\;\; f(x) < f(y) \]
Siccome $\{a_i\}_{i\in\omega}$ enumera~$A$, possiamo scrivere $x=a_m$
e~$y=a_n$, per opportuni $m,n$. Fissato $t>m,n$ abbiamo $x,y\in\Dom(f_t)$,
per cui, sfruttando il fatto che~$f_t$ \`e un isomorfismo di ordini
		\[ x < y \;\;\leftrightarrow\;\; f(x) = f_t(x) < f_t(y) = f(y)
			\]

Mostriamo ora la costruzione, per ricorsione, di una successione
$\{f_i\}_{i \in \omega}$ che soddisfa le condizioni (1)--(4).
Poniamo, per cominciare, $f_0 = \emptyset$.

Per costruire $f_{i+1}$ passiamo attraverso un passo intermedio~$\drkblu
f_{i+0.5}:$
	\begin{itemize}
		\item se $a_i \in \Dom(f_i)$: allora $f_{i + 0.5} = f_i$.
		\item altrimenti: detto $\overline{j} =
\min\{j \in \omega | f_i \cup \{(a_i,q_j)\} \; \text{\`e un
isomorfismo}\}$\\ poniamo $f_{i + 0.5} = f_i \cup \{(a_i,q_{\ol j})\}$.
	\end{itemize}

Ora costriamo $\drkblu f_{i+1}:$
	\begin{itemize}
		\item se $q_i \in \Imm(f_{i + 0.5})$: allora $f_{i + 1} = f_{i + 0.5}$.
		\item altrimenti: detto $\ol l =
\min\{l \in \omega | f_{i + 0.5} \cup (a_{\ol l},q_i) \; \text{\`e
un isomorfismo}\}$\\
poniamo $f_{i + 1} = f_{i+0.5} \cup \{(a_{\ol l},q_i)\}$
	\end{itemize}
	Le propriet\`a (1)\ldots(4) seguono facilmente per induzione, a patto che la costruzione sia ben posta, ossia i minimi esistano. 
	Ad essere precisi, occorre dimostrare, per induzione su $i$, la seguente proposizione:
	\[ \forall i \in \omega \; \text{``la costruzione di $f_i$ \`e ben
posta e valgono (1)\ldots(4)''}
		\]
Assumiamo $a_i\not\in\Dom(f_i)$
	e verifichiamo che esiste
\[
	\overline{j}\;\;=\;\;\min\{j \in \omega | f_i \cup \{(a_i,q_j)\}
\]
l'analoga verifica per $\ol l$ \`e simmetrica.
	Per ipotesi induttiva, $A_i$ \`e finito. Detto $n = |A_i|$,
per l'osservazione~\ref{ord-fin}
	\[ A_i = \{\alpha_0,\ldots,\alpha_{n-1}\} \qquad\text{con} \qquad\alpha_0<\ldots<\alpha_{n-1}
		\]
	Ora, o $a_i <\alpha_0$, o $\alpha_k < a_i < \alpha_{k+1}$ per
	qualche $k$, o $\alpha_{n-1}<a_i$. Nel primo e terzo caso,
rispettivamente, siccome $\QQ$ non ha estremi, c'\`e un $q_j <
f_i(\alpha_0)$, o, rispettivamente, un $q_j > f_i(\alpha_n)$. Questo basta
perch\'e $f_i \cup \{(a_i,q_j)\}$ sia un isomorfismo.
	Nel secondo caso, per la densit\`a di $\QQ$, esiste $q_j$ con
$f_i(\alpha_k)<q_j<f_i(\alpha_{k+1})$ e, di nuovo, questo \`e sufficiente
perch\'e $f_i  \cup \{(a_i,q_j)\}$ sia un isomorfismo.
\end{proof}

\begin{corollary}\label{cor-emb-ord-num}
	Sia $(A,<)$ un ordine totale con $|A| \leq \aleph_0$. Allora esiste $B \subseteq \QQ$ tale che $(A,<) \sim (B,<)$ con l'ordinamento indotto su $B$ da $\QQ$.
\end{corollary}

	Volendo, si potrebbe dimostrare questo corollario ripetendo, con qualche variazione, la dimostrazione del teorema. Ora daremo, per\`o,
	un argomento che, invece, applica il teorema. \`E comodo definire, prima, il prodotto di ordini.
\end{note}

\begin{definition}[Prodotto lessicografico di ordini]
	Dati $(A,<_A)$ e $(B,<_B)$ definiamo ${\drkblu (A,<_A) \times
(B,<_B)}\; \Mydef\; (A \times B, <_{A \times B})$\\
	dove ${\drkblu (a,b) <_{A \times B} (a',b')}\; \Mydef\; (b <_B b') \lor (b = b' \land a <_A a')$.

{\drkgrn Ossia $(A,<_A) \times (B,<_B)$ \`e il prodotto cartesiano $A \times B$ munito dell'ordine che confronta prima la seconda componente.}
\end{definition}

\vfill\pagebreak

Visualmente, si pu\`o immaginare il prodotto lessicografico $(A,<_A) \times (B,<_B)$ come ``$(A,<_A)$ ripetuto $(B,<_B)$ volte''.

	\centering
	\includegraphics[width = \textwidth]{immagini/ordine_lessicografico.png}
\end{figure}

\begin{remark}
	Il prodotto \`e un ordine. Inoltre, se $(A, <_A)$ e $(B,<_B)$ sono ordini totali, allora anche $(A,<_A) \times (B,<_B)$ lo \`e.
\end{remark}

\begin{exercise}
	Dati $(A,<_A),(B,<_B)$ e $(C,<_C)$ dimostra che:
	\[ ((A,<_A) \times (B,<_B)) \times (C,<_C) \sim (A,<_A) \times ((B,<_B) \times (C,<_C))
		\]
	ossia che il prodotto lessicografico di ordini \`e associativo a meno di isomorfismi.
\end{exercise}

\begin{proof}[Dimostrazione di~\ref{cor-emb-ord-num}]
	Il caso $A = \emptyset$ \`e immediato. Supponiamo quindi $A \ne \emptyset$ e consideriamo:
	\[ (\QQ\times A,<) \;\;\Mydef\;\; (\QQ,<) \times (A,<)
		\]
	\begin{figure}[H]
		\centering
		\includegraphics[width = \textwidth]{immagini/coroll_iso_cantor.png}
	\end{figure}
	L'insieme $\QQ \times A$ \`e numerabile, inoltre, dato $(q,a) \in \QQ \times A$ abbiamo
	\[ (q-1,a) < (q,a) < (q+1,a)\]
	quindi $\QQ \times A$ non ha estremi. Per verificare che \`e
denso, fissiamo $(q_1,a_1) < (q_2, a_2)$.\\
Se $a_1 < a_2$, allora $(q_1,a_1) < (q_1+1,a_1) < (q_2,a_2)$.\\
Se $a_1 = a_2$, allora $(q_1,a_1) < \left(\frac{q_1+q_2}{2},a_1\right) < (q_2,a_2) = (q_2,a_1)$.


	Per il \hyperref[isoCantor]{teorema di isomorfismo di Cantor,}
$(\QQ\times A,<) \sim (\QQ,<)$.
	Si conclude osservando che $A$ \`e isomorfo a $\{0\}\times
A\subseteq \QQ \times A$.
\end{proof}

\begin{exercise}
	Dimostra che se $(A,<)$ \`e denso e $2 \leq |A| \leq \aleph_0$,
allora $(A,<)$ \`e isomorfo a uno dei seguenti intervalli di $\QQ$:
$\;[0,1]_{\QQ} \quad ]0,1]_{\QQ} \quad [0,1[_{\QQ} \quad ]0,1[_{\QQ}$.
\end{exercise}


\subsection{Il grafo random}

La tecnica di estendere indefinitamente isomorfismi parziali che ci ha permesso di dimostrare il teorema di isomorfismo di Cantor
si chiama \vocab{back-and-forth}, ed \`e un metodo fondamentale per
trovare isomorfismi fra strutture in teoria dei modelli.
Cogliamo questa occasione per suggerire un esercizio di applicazione della
medesima tecnica che \`e un po' complicato. Si tratta della costruzione
del \vocab{grafo random} o \vocab{grafo di Rado}.

\begin{definition}[Grafo]
	Un \vocab{grafo} $(V,e)$ \`e il dato di un insieme \textit{di
vertici} $V$ e di una relazione \textit{di adiacenza} $e\subseteq V\times
V$ simmetrica {\drkgrn ($(x,y) \in e \leftrightarrow (y,x) \in e$)} e 
	irriflessiva {\drkgrn ($(x,x) \not\in e$).}

{\drkgrn L'idea \`e che la presenza di un arco fra $x$ e $y$ \`e
rappresentata da $(x,y) \in e$. Gli archi non sono diretti. Fra due
vertici non pu\`o esserci pi\`u di un arco. Non ci sono archi che
connettono un vertice a se stesso.}
\end{definition}

Partiamo da un'idea intuitiva -- chi ha gi\`a seguito un corso di probabilit\`a sapr\`a formalizzare questa cosa in termini precisi. Data una probabilit\`a $p \in \, ]0,1[$ costruiamo un grafo
$G_p$ con insieme di vertici $\omega$ come segue. Per ogni coppia $(i,j) \in \omega \times \omega$ con $i < j$
lanciamo una moneta {\drkred che fa testa con probabilit\`a $p$} -- tutte
queste monete indipendentemente -- e, se viene testa, mettiamo un arco fra $i$ e $j$.

Potremmo pensare che i grafi $G_{0.01}$ e $G_{0.99}$ debbano venire molto
diversi: uno ha l'$1\%$ degli archi possibili, l'altro ha il $99\%$, insomma uno \`e quasi vuoto, l'altro quasi completo. Avviene,
tuttavia, che, con probabilit\`a~$1$, questi grafi sono isomorfi.

\begin{definition}[Isomorfismo fra grafi]
	I grafi $(V_1,e_1)$ e $(V_2,e_2)$ sono \vocab{isomorfi} se esiste
una bigezione $f\colon V_1 \to V_2$ tale che:
	\[ \forall {v,w \in V_1} \; (v,w) \in e_1 \leftrightarrow (f(v),f(w)) \in e_2
		\]
\end{definition}

Vediamo perch\'e. Dati due sottoinsiemi finiti $X$ e $Y$ di $\omega$, e
dato un vertice $v \not \in X \cup Y$ la probabilit\`a che $v$ sia connesso da un arco a tutti i vertici di $X$ e a nessuno di quelli di $Y$
\`e $p^{|X|}(1 - p)^{|Y|} > 0$. Avendo preso $X$ e $Y$ finiti, ci sono
infiniti vertici $v \not \in X \cup Y$.
Si capisce, quindi, che, con probabilit\`a~$1$, qualcuno di questi
vertici~$v$ sar\`a connesso 
a tutti gli $X$ e a nessuno degli $Y$. Usiamo l'esistenza di questo $v$
per definire formalmente {\drkred un} grafo random.

\begin{definition}[Grafo random]
	Il grafo $(V,e)$ \`e un grafo random se $|V|=\aleph_0$ e
	\[ \forall {X\in\psf(V)}\;\forall {Y\in\psf(V\setminus X)}\;
\exists {v \in V \setminus(X \cup Y)} \; \underbrace{X \times \{v\}
\subseteq e}_{{\drkgrn \forall {x\in X} \; (x,v) \in e}} \land
\underbrace{(Y \times \{v\}) \cap e = \emptyset}_{{\drkgrn\nexists {y \in
Y} \; (y,v) \in e}}
		\]
\end{definition}

Le considerazioni di sopra dimostrano che un qualunque~$G_p$, con
probabilit\`a~$1$, \`e un grafo random.

\begin{exercise}[Esistenza e unicit\`a del grafo random]
	Dimostra che esiste un grafo random, ed \`e unico a meno di
isomorfismi.
\end{exercise}
